const sleepUI={loading:false,owner:null};
const sleepStateLabel={recording:'Se înregistrează',awaiting_finish:'Așteaptă starea telefonului',awaiting_upload:'Așteaptă înregistrările',analyzing:'Se analizează',needs_retry:'Necesită reîncercare',complete:'Raport disponibil'};
function clearSleepReports(){sleepUI.loading=false;sleepUI.owner=null;$('sleep-reports').querySelectorAll('.file-card').forEach(releaseCard);$('sleep-reports').replaceChildren();}
async function refreshSleepReports(){
 if(!auth||sleepUI.loading)return;const current=auth;sleepUI.loading=true;sleepUI.owner=current;
 try{const {sessions}=await api('/v2/sleep/sessions');if(auth!==current)return;const root=$('sleep-reports');root.querySelectorAll(':scope>.empty').forEach(n=>n.remove());const ids=new Set(sessions.map(s=>s.id));for(const old of root.querySelectorAll(':scope>details'))if(!ids.has(old.dataset.sleep)){old.querySelectorAll('.file-card').forEach(releaseCard);old.remove();}
 if(!sessions.length)empty(root,'Raportul începe cu o înregistrare.','Fragmentele se trimit automat cât timp înregistrezi.');
 for(const s of sessions){let card=[...root.querySelectorAll(':scope>details')].find(d=>d.dataset.sleep===s.id);if(!card){card=el('details',undefined,'section-disclosure');card.dataset.sleep=s.id;const summary=el('summary');summary.append(el('strong'),el('span',undefined,'muted'));const body=el('div',undefined,'sleep-report-body');card.append(summary,body);card.addEventListener('toggle',()=>{if(card.open&&!card.dataset.loaded){card.dataset.loaded='yes';openSleepReport(s.id,body);}});root.append(card);}card.querySelector('summary strong').textContent=dt(s.started_at);card.querySelector('summary span').textContent=sleepStateLabel[s.state]||s.state;const signature=JSON.stringify([s.state,s.chunk_count,s.analyzed_ms,s.analysis,s.snoring]);const playing=[...card.querySelectorAll('audio')].some(a=>!a.paused&&!a.ended);if(card.dataset.signature!==signature&&!playing){card.dataset.signature=signature;if(card.open)openSleepReport(s.id,card.querySelector('.sleep-report-body'));else delete card.dataset.loaded;}}
 }catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}finally{if(sleepUI.owner===current)sleepUI.loading=false;}
}
async function openSleepReport(id,body,cursor=null){
 const current=auth,epoch=cursor?body.dataset.reportEpoch:String((Number(body.dataset.reportEpoch)||0)+1);if(!cursor)body.dataset.reportEpoch=epoch;
 try{const r=await api('/v2/sleep/sessions/'+id+(cursor?'?cursor='+encodeURIComponent(cursor):''));if(auth!==current||!body.isConnected||body.dataset.reportEpoch!==epoch)return;
 if(!cursor){body.querySelectorAll('.file-card').forEach(releaseCard);body.replaceChildren();const head=el('div',undefined,'row');head.append(el('span',duration(r.recorded_ms)+' primit · '+duration(r.analyzed_ms)+' analizat','pill'),button('Actualizează raportul',()=>openSleepReport(id,body)));body.append(head);body.append(el('p',r.analysis_consent?'Transcriere automată · verifică prin ascultare.':'Analiza AI a fost oprită pentru această sesiune.','small muted'));body.append(el('p',r.snoring?.status==='unavailable'?'Analiza sunetelor nu este disponibilă.':(r.snoring.possible_intervals?r.snoring.possible_intervals+' intervale cu posibil sforăit':'Fără sforăit detectat în audio-ul procesat')+' · '+duration(r.snoring.analyzed_ms)+' analizat pe telefon.','small muted'));const help=el('a','Despre analiza audio','inline-help');help.href='#privacy-sleep';body.append(help);if(!r.chunks.length)empty(body,'Încă nu au ajuns fragmentele.','Raportul se completează după upload.');}
 body.querySelector('.sleep-more')?.remove();
 for(const c of r.chunks){const card=el('article',undefined,'file-card');card.style.marginTop='18px';card.append(el('h3',dt(c.recorded_from)+' · '+duration(c.duration_ms)));const stateLabel={pending:'În așteptare',analyzing:'Se analizează',complete:'Analizat',partial:'Analiză parțială',failed:'Analiza nu a reușit',unavailable:'Analiza nu este disponibilă',rejected:'Fișierul nu a putut fi analizat',skipped:'Fără analiză AI'};card.append(el('p',stateLabel[c.state]||c.state,'small muted'));
 card.append(button('Ascultă fragmentul',async b=>{const blob=await api('/v2/sessions/'+c.id+'/items/'+c.item_id,{},true);if(auth!==current||!card.isConnected||body.dataset.reportEpoch!==epoch)return;const u=URL.createObjectURL(new Blob([blob],{type:'audio/mp4'}));urls.push(u);card.dataset.mediaUrl=u;const audio=el('audio');audio.src=u;audio.controls=true;audio.preload='metadata';audio.setAttribute('aria-label','Înregistrare '+dt(c.recorded_from));card.append(audio);b.remove();try{await audio.play();}catch{}}));
 if(c.acoustic?.status==='complete'&&c.acoustic.events.length)card.append(el('p','Posibil sforăit · '+c.acoustic.events.map(e=>duration(e.start_ms)+'–'+duration(e.end_ms)).join(', '),'small muted'));
 if(c.acoustic?.status==='unavailable')card.append(el('p','Analiza locală a sunetelor nu a fost disponibilă.','small muted'));
 if(c.result?.transcript){const quote=el('blockquote',c.result.transcript);quote.style.cssText='margin:16px 0;padding-left:14px;border-left:2px solid var(--accent);white-space:pre-wrap;overflow-wrap:anywhere';card.append(quote);if(c.result.topics?.length){const topics=el('div',undefined,'row');for(const t of c.result.topics)topics.append(el('span',t.title,'pill'));card.append(topics);}}
 else if(c.result?.transcript_status==='empty')card.append(el('p','Nu s-a distins vorbire în transcrierea automată.','small muted'));
 if(c.result?.error?.message)card.append(el('p',c.result.error.message,'small muted'));
 if(['failed','unavailable','pending'].includes(c.state)&&r.analysis_consent)card.append(button('Reîncearcă analiza',async()=>{await api('/v2/sleep/sessions/'+id+'/chunks',{method:'POST',body:JSON.stringify({recording_session_id:c.id})});await openSleepReport(id,body);await refreshSleepReports();}));
 if(c.result?.limitations?.length){const d=el('details',undefined,'raw');d.append(el('summary','Limitele transcrierii'),el('p','Textul poate conține erori. Nu identifică vorbitorul, starea de somn sau o afecțiune.'));if(c.result.transcript_status!=='empty'&&c.result.limitations.some(v=>v.includes('truncat')||v.includes('limited')))d.append(el('p','Textul afișat este parțial.'));card.append(d);}
 body.append(card);}
 if(r.next_cursor)body.append(button('Mai multe fragmente',()=>openSleepReport(id,body,r.next_cursor),'sleep-more'));
 if(!cursor){const remove=button('Șterge raportul',async()=>{if(!confirm('Ștergi raportul și transcrierile? Fișierele audio rămân până la expirare sau ștergerea lor din Sesiuni.'))return;await api('/v2/sleep/sessions/'+id,{method:'DELETE'});await refreshSleepReports();},'small subtle');body.append(remove);}
 }catch(e){if(auth===current&&body.isConnected&&body.dataset.reportEpoch===epoch&&e.name!=='AbortError'){body.append(el('p',e.message,'notice'));}}
}
$('sleep-refresh').addEventListener('click',refreshSleepReports);
