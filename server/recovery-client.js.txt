const recoveryUI={data:null,layer:null,loading:false,inflight:null,markers:new Map(),durations:new Map(),pending:new Set()};
const recoveryCall=(path,body,method)=>api('/v2/recovery/'+path,{method:method||(body?'POST':'GET'),...(body?{body:JSON.stringify(body)}:{})});
function clearRecovery(){recoveryUI.data=null;recoveryUI.loading=false;recoveryUI.inflight=null;recoveryUI.layer?.clearLayers();recoveryUI.layer?.remove();recoveryUI.layer=null;recoveryUI.markers.clear();recoveryUI.durations.clear();recoveryUI.pending.clear();$('recovery-devices')?.replaceChildren();$('recovery-error').textContent='';}
async function refreshRecovery(){if(recoveryUI.loading)return recoveryUI.inflight;if(!auth||$('page-social').hidden||document.hidden)return;const current=auth;recoveryUI.loading=true;recoveryUI.inflight=(async()=>{try{const result=await recoveryCall('devices');if(auth!==current)return;recoveryUI.data=result.devices;renderRecovery();$('recovery-error').textContent='';}catch(e){if(auth===current&&e.name!=='AbortError'){$('recovery-error').textContent='Actualizarea telefoanelor a eșuat: '+e.message;renderRecovery();}}finally{if(auth===current){recoveryUI.loading=false;recoveryUI.inflight=null;}}})();return recoveryUI.inflight;}
function recoveryMarkers(){
 if(!socialUI.map||!recoveryUI.data)return;if(!recoveryUI.layer)recoveryUI.layer=L.layerGroup().addTo(socialUI.map);const ids=new Set();
 for(const d of recoveryUI.data){const p=d.position;if(!p||p.at<=Date.now()-86400000)continue;ids.add(d.id);const fresh=socialFresh(p);let entry=recoveryUI.markers.get(d.id);
  const badge=el('span','⌖','social-pin'+(fresh?'':' recovery-stale'));badge.title=d.name;const icon=L.divIcon({html:badge,className:'social-marker',iconSize:[48,52],iconAnchor:[24,48]});
  if(!entry){entry={marker:L.marker([p.lat,p.lon],{title:d.name,alt:d.name,icon}).addTo(recoveryUI.layer),circle:L.circle([p.lat,p.lon],{radius:Math.max(1,p.accuracy),color:fresh?'#426b48':'#89928c',fillOpacity:.1,weight:1}).addTo(recoveryUI.layer)};recoveryUI.markers.set(d.id,entry);}else{entry.marker.setLatLng([p.lat,p.lon]);entry.marker.setIcon(icon);entry.circle.setLatLng([p.lat,p.lon]);entry.circle.setRadius(Math.max(1,p.accuracy));entry.circle.setStyle({color:fresh?'#426b48':'#89928c'});}
  const popup=el('div',undefined,'social-map-popup');popup.append(el('strong',d.name),el('p',(fresh?'Poziție recentă':'Ultima poziție cunoscută')+' · '+socialAge(p.at),'social-meta'),el('p','±'+Math.round(p.accuracy)+' m · '+p.battery+'% baterie','social-meta'),socialNav(p));if(entry.marker.getPopup())entry.marker.setPopupContent(popup);else entry.marker.bindPopup(popup);
 }
 for(const[id,entry]of recoveryUI.markers)if(!ids.has(id)){recoveryUI.layer.removeLayer(entry.marker);recoveryUI.layer.removeLayer(entry.circle);recoveryUI.markers.delete(id);}
 if($('social-map-fit'))$('social-map-fit').disabled=!ids.size&&!socialUI.markers.size;if(!socialUI.fitted&&ids.size)socialFit();
}
async function recoveryAction(path,body,method,trigger,deviceId){const current=auth;if(deviceId&&recoveryUI.pending.has(deviceId))return;if(deviceId)recoveryUI.pending.add(deviceId);if(trigger)trigger.disabled=true;try{await recoveryCall(path,body,method);if(auth===current){if(recoveryUI.loading)await recoveryUI.inflight;if(auth===current)await refreshRecovery();}}catch(e){if(auth===current&&e.name!=='AbortError')$('recovery-error').textContent=e.message;}finally{if(auth===current){if(deviceId)recoveryUI.pending.delete(deviceId);if(trigger)trigger.disabled=false;renderRecovery();}}}
function renderRecovery(){
 const box=$('recovery-devices'),openMenus=new Set([...box.querySelectorAll('[data-social-menu][open]')].map(n=>n.dataset.socialMenu));box.replaceChildren();if(!recoveryUI.data?.length)socialEmpty(box,'⌖','Activează Găsire telefon în aplicație.');
 for(const d of recoveryUI.data||[]){const online=d.online&&d.seen_at>Date.now()-90000,card=el('article',undefined,'social-card'),head=el('div',undefined,'social-person');head.append(el('span','⌖','social-avatar'),el('strong',d.name));card.append(head,el('p',online?'Conectat':'Offline','recovery-status'+(online?' is-online':'')));
  const statuses={location_off:'GPS oprit pe telefon.',permission_missing:'Permite locația pe telefon.',notification_missing:'Permite notificările pe telefon.',expired:'Cererea a expirat.',stopped:'Căutare oprită.'};if(statuses[d.status])card.append(el('p',statuses[d.status],'social-status'));
  const actions=el('div',undefined,'social-actions'),more=socialMore('recovery-'+d.id,'Opțiuni telefon'),p=d.position,hasPosition=p&&p.at>Date.now()-86400000;
  if(d.command){card.append(el('p',d.command.phase==='active'?'Căutare activă':'Așteptăm confirmarea telefonului','social-status'));more.append(el('p','Până la '+dt(d.command.until),'social-meta'));const stop=button('Oprește căutarea',()=>recoveryAction('devices/'+d.id+'/command?id='+encodeURIComponent(d.command.id),null,'DELETE',stop,d.id));stop.disabled=recoveryUI.pending.has(d.id);actions.append(stop);}
  else{const minutes=el('select');minutes.setAttribute('aria-label','Durata căutării pentru '+d.name);for(const n of [5,15,30]){const o=el('option',n+' minute');o.value=String(n);o.selected=n===(recoveryUI.durations.get(d.id)||15);minutes.append(o);}minutes.addEventListener('change',()=>recoveryUI.durations.set(d.id,Number(minutes.value)));more.append(minutes);const start=button('Localizează',()=>recoveryAction('devices/'+d.id+'/command',{id:crypto.randomUUID(),minutes:Number(minutes.value)},'POST',start,d.id),'primary');start.disabled=recoveryUI.pending.has(d.id);actions.append(start);if(!online)more.append(el('p','Cererea așteaptă reconectarea până la 5 minute.','social-meta'));}
  if(hasPosition){card.append(el('p',(socialFresh(p)?'Poziție recentă':'Ultima poziție cunoscută')+' · '+socialAge(p.at),'social-meta'));actions.append(button('Pe hartă',()=>{socialShowMap(p,p.accuracy>1000?12:16);recoveryUI.markers.get(d.id)?.marker.openPopup();}));more.append(el('p',`±${Math.round(p.accuracy)} m · ${p.battery}% baterie`,'social-meta'),el('p',`${p.lat.toFixed(5)}, ${p.lon.toFixed(5)} · ${dt(p.at)}`,'social-meta'),socialNav(p));}else card.append(el('p','Nicio poziție în ultimele 24 h','social-meta'));
  more.append(el('p',d.seen_at?'Ultimul contact: '+dt(d.seen_at):'Încă nu s-a conectat.','social-meta'));const forget=button('Dezactivează găsirea',()=>recoveryAction('devices/'+d.id+'/grant',null,'DELETE',forget,d.id));forget.disabled=recoveryUI.pending.has(d.id);more.append(forget);if(openMenus.has(more.dataset.socialMenu))more.open=true;card.append(actions,more);box.append(card);
 }
 recoveryMarkers();
}
