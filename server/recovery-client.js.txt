const recoveryUI={data:null,layer:null,loading:false};
const recoveryCall=(path,body,method)=>api('/v2/recovery/'+path,{method:method||(body?'POST':'GET'),...(body?{body:JSON.stringify(body)}:{})});
function clearRecovery(){recoveryUI.data=null;recoveryUI.loading=false;recoveryUI.layer?.clearLayers();recoveryUI.layer?.remove();recoveryUI.layer=null;$('recovery-devices')?.replaceChildren();$('recovery-error').textContent='';}
async function refreshRecovery(){if(!auth||$('page-social').hidden||document.hidden||recoveryUI.loading)return;const current=auth;recoveryUI.loading=true;try{const result=await recoveryCall('devices');if(auth!==current)return;recoveryUI.data=result.devices;renderRecovery();$('recovery-error').textContent='';}catch(e){if(auth===current&&e.name!=='AbortError')$('recovery-error').textContent='Nu pot actualiza starea telefoanelor: '+e.message;}finally{if(auth===current)recoveryUI.loading=false;}}
function recoveryMarkers(){if(!socialUI.map||!recoveryUI.data)return;if(!recoveryUI.layer)recoveryUI.layer=L.layerGroup().addTo(socialUI.map);recoveryUI.layer.clearLayers();for(const d of recoveryUI.data){const p=d.position;if(!p||p.at<=Date.now()-86400000)continue;const fresh=p.at>Date.now()-120000;const badge=el('span','📱','social-pin'+(fresh?'':' recovery-stale'));badge.title=d.name;const popup=el('div');popup.append(el('strong',d.name),el('p',(fresh?'Poziție recentă':'Ultima poziție cunoscută')+' · '+dt(p.at)+' · ±'+Math.round(p.accuracy)+' m'));L.marker([p.lat,p.lon],{icon:L.divIcon({html:badge,className:'social-marker',iconSize:[48,52],iconAnchor:[24,48]})}).addTo(recoveryUI.layer).bindPopup(popup);L.circle([p.lat,p.lon],{radius:Math.max(1,p.accuracy),color:fresh?'#d3ebab':'#89928c',fillOpacity:.12,weight:1}).addTo(recoveryUI.layer);}}
async function recoveryAction(path,body,method,trigger){const current=auth;if(trigger)trigger.disabled=true;try{await recoveryCall(path,body,method);if(auth===current)await refreshRecovery();}catch(e){if(auth===current&&e.name!=='AbortError')$('recovery-error').textContent=e.message;}finally{if(trigger&&auth===current)trigger.disabled=false;}}
function renderRecovery(){const box=$('recovery-devices');box.replaceChildren();if(!recoveryUI.data?.length)box.append(el('p','Activează mai întâi Găsirea din site pe telefon: Hartă → Telefon pierdut. Activarea trebuie făcută înainte să pierzi telefonul.','muted'));
 for(const d of recoveryUI.data||[]){const card=el('article',undefined,'social-card');card.append(el('strong',d.name),el('p',d.online?'Telefonul răspunde':'Telefonul nu răspunde momentan','muted small'),el('p',d.seen_at?'Ultimul contact: '+dt(d.seen_at):'Încă nu s-a conectat.','muted small'));
 const statuses={location_off:'Locația Android este dezactivată. Site-ul nu poate porni GPS-ul.',permission_missing:'Permisiunea de locație lipsește pe telefon.',notification_missing:'Notificările trebuie permise pe telefon.',expired:'Cererea a expirat.',stopped:'Căutarea a fost oprită.'};if(statuses[d.status])card.append(el('p',statuses[d.status]));
 if(d.command){card.append(el('p',d.command.phase==='active'?'Telefonul a confirmat pornirea căutării.':'Cerere în așteptare; încă nu avem confirmarea telefonului.'),el('p','Căutare până la '+dt(d.command.until),'muted small'));const stop=button('Oprește căutarea',()=>recoveryAction('devices/'+d.id+'/command?id='+encodeURIComponent(d.command.id),null,'DELETE',stop));card.append(stop);}
 else{const minutes=el('select');minutes.setAttribute('aria-label','Durata căutării pentru '+d.name);for(const n of [5,15,30]){const o=el('option',n+' minute');o.value=String(n);o.selected=n===15;minutes.append(o);}const start=button('Localizează telefonul',()=>recoveryAction('devices/'+d.id+'/command',{id:crypto.randomUUID(),minutes:Number(minutes.value)},'POST',start));card.append(minutes,start);if(!d.online)card.append(el('p','Cererea poate aștepta reconectarea până la cinci minute. Un telefon oprit sau o aplicație oprită forțat nu poate răspunde.','muted small'));}
 const p=d.position;if(p&&p.at>Date.now()-86400000){const fresh=p.at>Date.now()-120000;card.append(el('p',(fresh?'Poziție recentă':'Ultima poziție cunoscută · nu este confirmată acum')+' · '+dt(p.at)),el('p',`Precizie estimată ±${Math.round(p.accuracy)} m · baterie ${p.battery}%`),el('p',`${p.lat.toFixed(5)}, ${p.lon.toFixed(5)}`,'muted small'),button('Arată telefonul pe hartă',()=>socialUI.map.flyTo([p.lat,p.lon],Math.max(10,p.accuracy>1000?12:16),{duration:.8})),socialNav(p));}else card.append(el('p','Nu există încă o poziție primită în ultimele 24 de ore.','muted small'));
 const forget=button('Dezactivează găsirea acestui telefon',()=>recoveryAction('devices/'+d.id+'/grant',null,'DELETE',forget));card.append(forget);box.append(card);
 }recoveryMarkers();}
