async function startJourney(){
 if(!navigator.geolocation){notice('Browserul nu oferă acces la locație. Folosește aplicația.');return;}
 const current=auth,id=crypto.randomUUID();try{const session=await socialCall('journey/session',{id,consent:true});if(auth!==current)return;journeyInvalidate();journeyUI.recording=id;let lastAt=0;
 journeyUI.watch=navigator.geolocation.watchPosition(p=>{
  if(auth!==current||journeyUI.recording!==id)return;
  const at=Math.floor(p.timestamp);if(p.coords.accuracy>50||at<session.started_at||at<Date.now()-90000||at>Date.now()+10000||at-lastAt<25000)return;
  if(journeyUI.queue.length>=2000){navigator.geolocation.clearWatch(journeyUI.watch);journeyUI.watch=null;notice('Explorarea locală s-a oprit. Reîncearcă sincronizarea înainte de a închide pagina.');return;}
  lastAt=at;journeyUI.queue.push({id:crypto.randomUUID(),session:id,points:[{at,lat:p.coords.latitude,lon:p.coords.longitude,accuracy:p.coords.accuracy,speed:Math.min(100,Math.max(0,p.coords.speed||0))}]});journeyFlush();

async function journeyFlush(){

  if(e.status===400){queue.shift();notice('Un punct invalid sau prea vechi a fost omis. Sincronizarea continuă.');continue;}
  if([403,404,409].includes(e.status)){navigator.geolocation?.clearWatch(journeyUI.watch);journeyUI.watch=null;journeyUI.recording=null;queue.length=0;notice('Explorarea a fost oprită sau accesul a fost revocat.');break;}throw e;
 }if(auth===current&&journeyUI.queue===queue)queue.shift();}}
 catch(e){if(auth===current&&e.name!=='AbortError')notice('Punctele așteaptă conexiunea. Păstrează această pagină deschisă.');}
 finally{if(journeyUI.queue===queue){journeyUI.sending=false;journeyUI.flush=null;}}})();journeyUI.flush=flush;await flush;if(journeyUI.flush===flush)journeyUI.flush=null;
