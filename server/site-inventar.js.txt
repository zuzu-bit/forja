  async list(box, params, more = false) {
    const st = Inventar.mirror, key = JSON.stringify(params);
    if (!more || st.key !== key) { st.key = key; st.items = []; st.next = null; st.total = 0; }
    const token = ++st.token;
    if (!more) fill(box, h('div', {class: 'mgrid'}, Array.from({length: 12}, () => h('div', {class: 'sk sk-thumb'}))));
    st.loading = true;
    try {
      const q = new URLSearchParams({limit: '60'});
      if (params.kind && params.kind !== 'all') q.set('kind', params.kind);
      if (params.q) q.set('q', params.q);
      if (params.group) q.set('group', params.group);
      if (params.album !== undefined) q.set('album', params.album);
      if (more && st.next) q.set('after', st.next);
      const d = await api('/v2/mirror?' + q);
      if (token !== st.token) return;
      st.items.push(...(d?.items || [])); st.next = d?.next || null; st.total = d?.total ?? st.items.length;
      Inventar.drawList(box, params);
    } catch (e) {
      if (e?.name === 'AbortError' || token !== st.token) return;
      fill(box, errorState(e?.message, () => Inventar.list(box, params)));
    } finally { if (token === st.token) st.loading = false; }
  },

/** Starea rulării, cum o scrie telefonul (InvSummaryDoc): în analiză, gata (așteaptă acordul), se aplică, aplicată, oprită. */
function runStateText(run) {
  const p = run.progress, pct = p && p.total ? Math.round(100 * Math.min(1, (p.done || 0) / p.total)) : null;
  switch (run.state) {
    case 'scanning': case 'grouping': case 'naming': return 'în analiză' + (pct !== null ? ' · ' + pct + '%' : '');
    case 'ready': return 'gata, așteaptă acordul';
    case 'applying': return 'se aplică' + (p?.total ? ' · ' + fmt.int(p.done) + ' din ' + fmt.int(p.total) : '');
    case 'failed': return 'oprită';
    default: return '';
  }
}
function runStateBadge(run) {
  const text = runStateText(run); if (!text) return run.state === 'done' ? badge('APLICAT', 'pos') : null;
  return h('span', {class: 'run-state'}, badge(text, run.state === 'failed' ? 'ember' : 'amber'));
}
function coverCollage(run, f) {
  const keys = (Array.isArray(f.covers) ? f.covers : []).slice(0, 4).filter(k => /^[A-Za-z0-9_-]{1,40}$/.test(k));
  if (!keys.length) return null;
  return h('span', {class: 'folder-covers'}, [0, 1, 2, 3].map(i => {
    const cell = h('span');
    if (keys[i] && Inventar.observer) { cell.dataset.id = keys[i]; cell.dataset.src = '/v2/mirror/cover/' + encodeURIComponent(run.id) + '/' + encodeURIComponent(keys[i]); Inventar.observer.observe(cell); }
    return cell;
  }));
}
function runCard(run, full) {
  const folders = [...(run.folders || [])].sort((a, b) => (b.count || 0) - (a.count || 0));
  const shown = full ? folders : folders.slice(0, 8), max = Math.max(1, ...folders.map(f => f.count || 0));
  const docs = run.kind === 'docs', trash = run.trash || {count: 0, bytes: 0};
  const reasons = Object.entries(trash.byReason || {}).filter(([, n]) => num(n) && n > 0).sort((a, b) => b[1] - a[1]);
  const fails = Object.entries(run.failures || {}).filter(([, n]) => num(n) && n > 0).sort((a, b) => b[1] - a[1]);
  const recover = num(trash.expiresAt) && trash.expiresAt > Date.now() ? 'se recuperează până la ' + fmt.day(trash.expiresAt) : num(trash.expiresAt) ? 'fereastra de recuperare s-a închis' : 'se recuperează 30 de zile';
  return h('article', {class: 'card run-card'},
    h('div', {class: 'run-head'},
      h('span', {class: 'run-kind'}, icon(docs ? 'doc' : 'photo', 20), monoLabel((docs ? 'DOCUMENTE' : 'POZE') + (run.scope?.label ? ' · ' + String(run.scope.label).toUpperCase() : ''))),
      h('span', {class: 'run-state'}, runStateBadge(run), monoLabel(fmt.dayLong(run.finishedAt).toUpperCase() + ' · ' + fmt.time(run.finishedAt)))),
    h('div', {class: 'run-stats' + (num(run.freedBytes) ? '' : ' three')},
      statBlock('DOSARE', fmt.int(folders.length)),
      statBlock('MUTATE', fmt.int(run.moved)),
      statBlock('DE ARUNCAT', fmt.int(trash.count), null, trash.bytes ? fmt.bytes(trash.bytes) : null),
      num(run.freedBytes) ? statBlock('ELIBERAT', fmt.bytes(run.freedBytes)) : null),
    h('div', {class: 'run-dest'}, icon('folder', 18), h('span', {class: 'run-dest-label', text: run.dest?.label || 'Destinație'}), run.dest?.path ? h('code', {class: 'run-dest-path', text: run.dest.path}) : null),
    run.provider ? monoLabel('NUMELE DOSARELOR · ' + String(run.provider).toUpperCase(), 'dim') : null,
    run.error ? h('p', {class: 'run-failed'}, icon('info', 16), run.error) : null,
    run.failed > 0 ? h('p', {class: 'run-failed'}, icon('info', 16), plural(run.failed, 'fișier nu s-a mutat', 'fișiere nu s-au mutat') + (fails.length ? ': ' + fails.map(([k, n]) => fmt.int(n) + ' ' + (FAIL_REASONS[k] || 'altele')).join(', ') : '') + '. Rămân la locul lor.') : null,
    reasons.length ? h('div', {class: 'run-reasons'}, reasons.map(([k, n]) => badge(fmt.int(n) + ' ' + (TRASH_REASONS[k] || k)))) : null,
    h('div', {class: 'folder-grid'},
      trash.count > 0 ? h('div', {class: 'folder trash'}, h('span', {class: 'folder-ic'}, icon('trash', 20)), h('span', {class: 'folder-name', text: 'De aruncat'}),
        h('span', {class: 'folder-meta', text: fmt.int(trash.count) + ' · ' + fmt.bytes(trash.bytes)}), h('span', {class: 'folder-note', text: docs ? 'în „De aruncat (FORJA)”' : recover})) : null,
      shown.map(f => h('div', {class: 'folder'}, coverCollage(run, f) || h('span', {class: 'folder-ic'}, icon('folder', 20)), h('span', {class: 'folder-name', text: f.name || 'Dosar'}),
        f.theme ? h('span', {class: 'folder-theme', text: f.theme}) : null,
        h('span', {class: 'folder-meta', text: fmt.int(f.count) + (num(f.bytes) ? ' · ' + fmt.bytes(f.bytes) : '')}), bar((f.count || 0) / max)))),
    !full && folders.length > shown.length ? h('a', {href: '#inventar/' + encodeURIComponent(run.id), class: 'panel-link'}, 'Toate cele ' + plural(folders.length, 'dosar', 'dosare'), icon('next', 16)) : null,
    full && run.appVersion ? monoLabel('FORJA ' + run.appVersion + ' · ÎNCEPUT ' + fmt.time(run.startedAt), 'dim') : null);
}
