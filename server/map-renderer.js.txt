  const fc = features => ({type: 'FeatureCollection', features});

  const geo = {
    empty: () => fc([]),
    /** Celulele din explore/state, cu `mode` normalizat (fără mod → pe jos, ca în aplicație). */
    cells(collection) {
      const out = [];
      for (const f of collection?.features || []) {
        if (f?.geometry?.type !== 'Polygon') continue;
        const p = f.properties || {};
        out.push({type: 'Feature', geometry: f.geometry, properties: {id: String(p.id || ''), visits: isNum(p.visits) ? p.visits : 1, mode: MODES.has(p.mode) ? p.mode : 'walk'}});

    },
    /** Doar laturile fără vecin cucerit (grila „x_y”), într-o MultiLineString — teritoriul, nu tabla de șah. */
    edges(cells) {
      const list = cells.features, keys = new Set(), parsed = [];
      for (const f of list) {
        const m = /^(-?\d+)_(-?\d+)$/.exec(f.properties.id || '');
        const x = m ? Number(m[1]) : null, y = m ? Number(m[2]) : null;
        parsed.push([x, y]); if (m) keys.add(x + '_' + y);
      }
      const lines = [];
      list.forEach((f, i) => {
        const ring = f.geometry.coordinates[0];
        let minLng = Infinity, minLat = Infinity, maxLng = -Infinity, maxLat = -Infinity;
        for (const [lng, lat] of ring) { minLng = Math.min(minLng, lng); maxLng = Math.max(maxLng, lng); minLat = Math.min(minLat, lat); maxLat = Math.max(maxLat, lat); }
        const [x, y] = parsed[i], known = x !== null;
        if (!known || !keys.has(x + '_' + (y + 1))) lines.push([[minLng, maxLat], [maxLng, maxLat]]);
        if (!known || !keys.has(x + '_' + (y - 1))) lines.push([[minLng, minLat], [maxLng, minLat]]);
        if (!known || !keys.has((x + 1) + '_' + y)) lines.push([[maxLng, minLat], [maxLng, maxLat]]);
        if (!known || !keys.has((x - 1) + '_' + y)) lines.push([[minLng, minLat], [minLng, maxLat]]);
      });
      return lines.length ? fc([{type: 'Feature', geometry: {type: 'MultiLineString', coordinates: lines}, properties: {}}]) : fc([]);
    },
    heat(cells) {
      return fc(cells.features.map(f => {
        const ring = f.geometry.coordinates[0];
        const lng = (ring[0][0] + ring[2][0]) / 2, lat = (ring[0][1] + ring[2][1]) / 2;
        return {type: 'Feature', geometry: {type: 'Point', coordinates: [lng, lat]}, properties: {visits: f.properties.visits}};
      }));
    },
    /** „lat,lng;lat,lng;…” → [[lat,lng],…]; perechile invalide se ignoră (MapGeo.parsePolyline). */
    parsePolyline(text) {
      const out = [];
      for (const pair of String(text || '').split(';')) {
        const i = pair.indexOf(','); if (i <= 0) continue;
        const lat = Number(pair.slice(0, i)), lng = Number(pair.slice(i + 1));
        if (isNum(lat) && isNum(lng) && Math.abs(lat) <= 90 && Math.abs(lng) <= 180) out.push([lat, lng]);
      }
      return out;
    },

    circle(lat, lng, radiusM, steps = 56) {
      const R = 6378137, pts = [], d = Math.max(1, radiusM) / R, la = lat * Math.PI / 180, lo = lng * Math.PI / 180;
      for (let i = 0; i <= steps; i++) {
        const b = (i / steps) * 2 * Math.PI;
        const lat2 = Math.asin(Math.sin(la) * Math.cos(d) + Math.cos(la) * Math.sin(d) * Math.cos(b));
        const lng2 = lo + Math.atan2(Math.sin(b) * Math.sin(d) * Math.cos(la), Math.cos(d) - Math.sin(la) * Math.sin(lat2));
        pts.push([lng2 * 180 / Math.PI, lat2 * 180 / Math.PI]);
      }
      return {type: 'Feature', geometry: {type: 'Polygon', coordinates: [pts]}, properties: {}};
    },

      return n ? [[a, b], [c, d]] : null;
    },
    coordsOf(collection) {

    }

  function distanceM(a, b) {
    const R = 6371000, rad = Math.PI / 180, dLat = (b[1] - a[1]) * rad, dLng = (b[0] - a[0]) * rad;
    const x = Math.sin(dLat / 2) ** 2 + Math.cos(a[1] * rad) * Math.cos(b[1] * rad) * Math.sin(dLng / 2) ** 2;
    return 2 * R * Math.asin(Math.min(1, Math.sqrt(x)));
  }

  const DPR = () => Math.max(2, Math.min(3, (root.devicePixelRatio || 1)));
  function canvas(w, h) {

  function starPath(g, cx, cy, R, r) {
    g.beginPath();
    for (let i = 0; i < 10; i++) { const a = -Math.PI / 2 + i * Math.PI / 5, rr = i % 2 ? r : R; g.lineTo(cx + Math.cos(a) * rr, cy + Math.sin(a) * rr); }
    g.closePath();
  }
  function drawPlace(stars, rec, selected) {

  function drawPill(text, tone = 'dark') {

  const SRC = ['forja-heat', 'forja-cells', 'forja-cells-edge', 'forja-streets', 'forja-focus', 'forja-accuracy', 'forja-places', 'forja-rec', 'forja-devices', 'forja-friends', 'forja-me', 'forja-pin'];
  const TOGGLES = {heat: ['forja-heat'], territories: ['forja-cells-fill', 'forja-cells-line'], streets: ['forja-streets-casing', 'forja-streets'], places: ['forja-halos', 'forja-places', 'forja-places-label', 'forja-rec', 'forja-rec-label'], friends: ['forja-friends', 'forja-friends-label']};
  const GASIRE_HIDDEN = ['forja-heat', 'forja-cells-fill', 'forja-cells-line', 'forja-streets-casing', 'forja-streets', 'forja-halos', 'forja-places', 'forja-places-label', 'forja-rec', 'forja-rec-label', 'forja-friends', 'forja-friends-label', 'forja-focus-glow', 'forja-focus'];
  const PICKABLE = ['forja-places', 'forja-rec', 'forja-friends', 'forja-devices', 'forja-me'];
  const HALO_PX_Z13 = 100 / (78271.517 * Math.cos(45 * Math.PI / 180) / 2 ** 13);

  function supported(gl) { return !!gl && typeof gl.Map === 'function'; }

  function create(container, opts = {}) {
    const gl = root.maplibregl;
    if (!supported(gl)) throw Error('Harta nu este disponibilă în acest browser.');
    const reduced = opts.reduced !== undefined ? !!opts.reduced : (() => { try { return root.matchMedia('(prefers-reduced-motion: reduce)').matches; } catch { return false; } })();
    const map = new gl.Map({container, style: opts.style || STYLE_URL, center: [25.0, 45.9], zoom: 5.6, maxPitch: 70, attributionControl: {compact: true}, fadeDuration: reduced ? 0 : 300, dragRotate: true, touchPitch: true});
    const state = {ready: false, failed: false, theme: opts.theme || 'night', mode: 'teren', threeD: false, layers: {heat: true, territories: true, streets: true, places: true, friends: true},
      raw: {}, fc: {}, images: new Map(), me: null, pulse: null, destroyed: false};
    const api = {map, get ready() { return state.ready; }, get threeD() { return state.threeD; }, get mode() { return state.mode; }};
    const duration = ms => reduced ? 0 : ms;
    let timeout = setTimeout(() => { if (!state.ready) fail(Error('Harta nu a răspuns la timp.'), false); }, opts.timeoutMs || 20000);

    function fail(error, fatal = true) {
      if (fatal) { state.ready = false; state.failed = true; }
      try { opts.onError && opts.onError(error, fatal); } catch { /* raportarea nu oprește harta */ }
    }
    map.on('error', e => {
      const err = e && e.error, msg = String(err && err.message || '');
      // Dale, glife, sprite sau o sursă a stilului: harta merge mai departe (MapLibre reîncearcă singur).
      if (state.ready || e.sourceId || e.tile || /glyph|sprite|image|tile|source/i.test(msg)) return;
      fail(err || Error('Stilul hărții nu s-a încărcat.'));
    });
    map.on('webglcontextlost', () => fail(Error('Placa grafică a oprit harta.')));
    map.on('webglcontextrestored', () => { if (state.failed) { state.failed = false; install(); } });
    map.on('styleimagemissing', e => provideImage(e.id));
    // Datele FORJA intră imediat ce stilul e gata („style.load”), nu după ultima dală („load”): o dală lentă de la
    // OpenFreeMap nu mai ține pe loc teritoriul, camarazii sau telefonul.
    map.on('style.load', () => { clearTimeout(timeout); install(); });
    map.on('load', () => { clearTimeout(timeout); install(); });
    // Pentru teste și capturi: containerul spune când harta a terminat de desenat ultimele date.
    map.on('idle', () => { try { container.dataset.idle = '1'; } catch { /* container detașat */ } });

    function install() {
      if (state.destroyed || state.ready) return;
      try {
        applyPalette(map, state.theme);
        const anchor = map.getLayer('building') ? 'building' : (map.getStyle().layers || []).find(l => l.type === 'symbol')?.id;
        for (const id of SRC) if (!map.getSource(id)) map.addSource(id, {type: 'geojson', data: EMPTY, ...(['forja-places', 'forja-me', 'forja-accuracy'].includes(id) ? {buffer: 0} : {})});
        const ground = (layer) => { if (!map.getLayer(layer.id)) map.addLayer(layer, anchor); };
        const top = (layer) => { if (!map.getLayer(layer.id)) map.addLayer(layer); };
        // Pragurile de zoom ale aplicației, coborâte cu un nivel: pe ecranul mare orașul întreg încape la ~12, iar acolo
        // vrem teritoriile, nu doar strălucirea.
        ground({id: 'forja-heat', type: 'heatmap', source: 'forja-heat', maxzoom: 13, paint: {
          'heatmap-weight': ['interpolate', ['linear'], ['to-number', ['get', 'visits']], 1, 0.35, 8, 1],
          'heatmap-radius': ['interpolate', ['linear'], ['zoom'], 6, 5, 10, 16, 13, 32], 'heatmap-intensity': 0.9,
          'heatmap-opacity': ['interpolate', ['linear'], ['zoom'], 10, 0.9, 11.5, 0.6, 12.6, 0],
          'heatmap-color': ['interpolate', ['linear'], ['heatmap-density'], 0, 'rgba(74,93,58,0)', 0.25, 'rgba(74,93,58,0.35)', 0.6, 'rgba(111,133,90,0.65)', 1, 'rgba(243,185,82,0.9)']}});
        ground({id: 'forja-cells-fill', type: 'fill', source: 'forja-cells', minzoom: 10.5, paint: {
          'fill-color': ['match', ['get', 'mode'], 'run', C.green, 'ride', C.teal, C.olive], 'fill-antialias': false,
          'fill-opacity': ['interpolate', ['linear'], ['zoom'], 10.5, 0, 12.2, 0.4]}});
        ground({id: 'forja-cells-line', type: 'line', source: 'forja-cells-edge', minzoom: 11, paint: {'line-color': C.olive2, 'line-width': 1, 'line-opacity': ['interpolate', ['linear'], ['zoom'], 11, 0, 12.5, 0.6]}});
        const width = base => ['interpolate', ['linear'], ['zoom'], 12, base * 0.45, 16, base, 20, base * 1.8];
        ground({id: 'forja-streets-casing', type: 'line', source: 'forja-streets', layout: {'line-cap': 'round', 'line-join': 'round'}, paint: {'line-color': '#1A1A1E', 'line-opacity': 0.5, 'line-width': width(7)}});
        ground({id: 'forja-streets', type: 'line', source: 'forja-streets', layout: {'line-cap': 'round', 'line-join': 'round'}, paint: {'line-color': C.amber, 'line-width': width(5), 'line-opacity': 0.92}});
        ground({id: 'forja-focus-glow', type: 'line', source: 'forja-focus', layout: {'line-cap': 'round', 'line-join': 'round'}, paint: {'line-color': C.amber, 'line-opacity': 0.25, 'line-width': width(18)}});
        ground({id: 'forja-focus', type: 'line', source: 'forja-focus', layout: {'line-cap': 'round', 'line-join': 'round'}, paint: {'line-color': '#FFE39A', 'line-width': width(6)}});
        ground({id: 'forja-halos', type: 'circle', source: 'forja-places', minzoom: 13, paint: {
          'circle-radius': ['interpolate', ['exponential', 2], ['zoom'], 13, HALO_PX_Z13, 19, HALO_PX_Z13 * 64],
          'circle-color': 'rgba(243,185,82,0.08)', 'circle-stroke-color': 'rgba(243,185,82,0.33)', 'circle-stroke-width': 1.5, 'circle-pitch-alignment': 'map'}});
        ground({id: 'forja-accuracy-fill', type: 'fill', source: 'forja-accuracy', paint: {'fill-color': C.amber, 'fill-opacity': 0.1}});
        ground({id: 'forja-accuracy-line', type: 'line', source: 'forja-accuracy', paint: {'line-color': C.amber, 'line-opacity': 0.55, 'line-width': 1.5}});
        const symbol = (id, source, extra = {}) => top({id, type: 'symbol', source, layout: {'icon-image': ['get', 'icon'], 'icon-anchor': 'bottom', 'icon-allow-overlap': true, 'icon-ignore-placement': true, 'symbol-sort-key': ['to-number', ['get', 'sort']], ...extra.layout}, paint: {'icon-opacity': ['to-number', ['coalesce', ['get', 'alpha'], 1]], ...extra.paint}, ...(extra.minzoom ? {minzoom: extra.minzoom} : {})});
        const label = (id, source, minzoom) => top({id, type: 'symbol', source, minzoom, layout: {'icon-image': ['get', 'label'], 'icon-anchor': 'top', 'icon-offset': [0, 3], 'icon-allow-overlap': false, 'icon-ignore-placement': false, 'icon-optional': false}, paint: {'icon-opacity': ['to-number', ['coalesce', ['get', 'alpha'], 1]]}});
        symbol('forja-places', 'forja-places'); label('forja-places-label', 'forja-places', 14.5);
        symbol('forja-rec', 'forja-rec'); label('forja-rec-label', 'forja-rec', 14.5);
        top({id: 'forja-me-pulse', type: 'circle', source: 'forja-me', paint: {'circle-color': C.olive2, 'circle-radius': 18, 'circle-opacity': 0.18, 'circle-pitch-alignment': 'map'}});
        symbol('forja-devices', 'forja-devices');
        symbol('forja-friends', 'forja-friends'); label('forja-friends-label', 'forja-friends', 0);
        symbol('forja-me', 'forja-me'); label('forja-me-label', 'forja-me', 0);
        symbol('forja-pin', 'forja-pin');
        if (!state.picking) for (const id of PICKABLE) {
          state.picking = true;
          map.on('click', id, e => { const f = e.features && e.features[0]; if (!f) return; const [lng, lat] = f.geometry.coordinates; opts.onPick && opts.onPick({kind: f.properties.kind, id: f.properties.id, lat, lng}); });
          map.on('mouseenter', id, () => { map.getCanvas().style.cursor = 'pointer'; });
          map.on('mouseleave', id, () => { map.getCanvas().style.cursor = ''; });
        }
        state.ready = true; state.failed = false;
        render(Object.keys(state.raw));
        applyVisibility();
        setBuildings();
        startPulse();
        opts.onReady && opts.onReady(api);
      } catch (e) { fail(e); }
    }

    function provideImage(id) {
      if (state.images.has(id) && map.hasImage && !map.hasImage(id)) try { const im = state.images.get(id); if (im) map.addImage(id, im.data, {pixelRatio: im.pixelRatio}); } catch { /* deja adăugată */ }
    }
    /** Înregistrează o imagine desenată o singură dată, după cheie. */
    function img(key, draw) {
      if (!state.images.has(key)) state.images.set(key, draw());
      if (state.ready) try { if (!map.hasImage || !map.hasImage(key)) { const im = state.images.get(key); if (im) map.addImage(key, im.data, {pixelRatio: im.pixelRatio}); } } catch { /* repetat */ }
      return key;
    }
    const pill = (text, tone) => img('pill|' + tone + '|' + text, () => drawPill(text, tone));

    function build(kind) {
      const r = state.raw;
      if (kind === 'cells') { const cells = geo.cells(r.cells); state.fc.cells = cells; return {'forja-cells': cells, 'forja-cells-edge': geo.edges(cells), 'forja-heat': geo.heat(cells)}; }
      if (kind === 'routes') { const streets = geo.streets(r.routes); state.fc.streets = streets; return {'forja-streets': streets}; }
      if (kind === 'focusRoute') return {'forja-focus': r.focusRoute ? geo.streets([r.focusRoute]) : EMPTY};
      if (kind === 'places') {
        const sel = r.selected || {}, out = [];
        for (const p of r.places || []) {
          const pt = pointOf(p.lat, isNum(p.lng) ? p.lng : p.lon); if (!pt) continue;
          const stars = Math.max(0, Math.min(5, Math.round(p.stars || 0))), selected = sel.kind === 'place' && sel.id === String(p.id);
          out.push({type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'place', id: String(p.id), sort: selected ? 3 : 1, icon: img(`place|${stars}|0|${selected ? 1 : 0}`, () => drawPlace(stars, false, selected)), label: pill(p.name ? String(p.name).slice(0, 26) : 'Loc fără nume', 'dark')}});
        }
        const places = fc(out); state.fc.places = places; return {'forja-places': places};
      }
      if (kind === 'recommended') {
        const sel = r.selected || {}, out = [];
        for (const p of r.recommended || []) {
          const pt = pointOf(p.lat, p.lng); if (!pt) continue;
          const stars = Math.max(0, Math.min(5, Math.round(p.stars || 0))), selected = sel.kind === 'rec' && sel.id === String(p.id);
          out.push({type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'rec', id: String(p.id), sort: selected ? 3 : 0, icon: img(`place|${stars}|1|${selected ? 1 : 0}`, () => drawPlace(stars, true, selected)), label: pill(String(p.name || 'Recomandare').slice(0, 26), 'dark')}});
        }
        return {'forja-rec': fc(out)};
      }
      if (kind === 'me') {
        const m = r.me, pt = m && pointOf(m.lat, m.lng); state.me = pt;
        if (!pt) return {'forja-me': EMPTY};
        const ghost = !!m.ghost, age = ago(m.at);
        return {'forja-me': fc([{type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'me', id: 'me', sort: 5, alpha: ghost ? 0.55 : 1, icon: img('av|me|' + (ghost ? 1 : 0), () => drawAvatar({me: true, ghost})), label: pill(ghost ? 'Tu · fantomă' : age && age !== 'acum' ? 'Tu · ' + age : 'Tu', 'dark')}}])};
      }
      if (kind === 'friends' || kind === 'family') {
        const sel = r.selected || {}, out = [], me = state.me;
        const add = (f, family) => {
          const pt = pointOf(f.lat, f.lng); if (!pt) return;
          const selected = sel.kind === 'friend' && sel.id === String(f.uid);
          const music = !family && !!(f.nowPlaying && f.nowPlaying.title);
          const key = `av|${initials(f.name)}|${f.state || ''}|${family ? 1 : 0}|${music ? 1 : 0}|${selected ? 1 : 0}`;
          const dist = me ? distanceM(me, pt) : null;
          const text = (String(f.name || '').split(/\s+/)[0] || 'Camarad') + (dist !== null ? ' · ' + km(dist) : f.at ? ' · ' + ago(f.at) : '');
          out.push({type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'friend', id: String(f.uid), sort: selected ? 4 : ['run', 'walk', 'ride'].includes(f.state) ? 2 : 1, alpha: family ? 0.55 : 1,
            icon: img(key, () => drawAvatar({name: f.name, state: f.state, family, music, selected})), label: pill(text, 'dark')}});
        };
        for (const f of r.friends || []) if (!f.ghost) add(f, false);
        const friendIds = new Set((r.friends || []).filter(f => !f.ghost && isNum(f.lat)).map(f => f.uid));
        for (const f of r.family || []) if (!friendIds.has(f.uid)) add(f, true);
        const friends = fc(out); state.fc.friends = friends; return {'forja-friends': friends};
      }
      if (kind === 'devices') {
        const out = [], acc = [];
        for (const d of r.devices || []) {
          // Cel mai nou punct: poziția din căutare rămâne 24 h, dar bătăile mută `last` mai departe.
          const p = d.position && (!d.last || d.position.at >= d.last.at) ? d.position : d.last; const pt = p && pointOf(p.lat, isNum(p.lon) ? p.lon : p.lng); if (!pt) continue;
          const fresh = !!(p === d.position && d.position.fresh) || (Date.now() - (p.at || 0) < 10 * 60000);
          out.push({type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'device', id: String(d.id), sort: 6, icon: img('dev|' + (fresh ? 1 : 0), () => drawDevice(fresh))}});
          if (isNum(p.accuracy) && p.accuracy > 0) acc.push(geo.circle(pt[1], pt[0], Math.min(p.accuracy, 5000)));
        }
        const devices = fc(out); state.fc.devices = devices; return {'forja-devices': devices, 'forja-accuracy': fc(acc)};
      }
      if (kind === 'pin') { const pt = r.pin && pointOf(r.pin.lat, r.pin.lng); return {'forja-pin': pt ? fc([{type: 'Feature', geometry: {type: 'Point', coordinates: pt}, properties: {kind: 'pin', id: 'pin', sort: 7, icon: img('pin', drawPin)}}]) : EMPTY}; }
      return {};
    }
    function render(kinds) {
      if (!state.ready) return;
      const order = ['me', 'cells', 'routes', 'focusRoute', 'places', 'recommended', 'friends', 'devices', 'pin'];
      const wanted = new Set(kinds.map(k => k === 'family' ? 'friends' : k === 'selected' ? ['places', 'recommended', 'friends'] : k).flat());
      if (wanted.has('me')) wanted.add('friends');
      for (const kind of order) {
        if (!wanted.has(kind)) continue;
        for (const [id, data] of Object.entries(build(kind))) { const src = map.getSource(id); if (src && src.setData) src.setData(data); }
      }
    }
    function visible(id) {
      if (state.mode === 'gasire') return !GASIRE_HIDDEN.includes(id);
      if (id === 'forja-accuracy-fill' || id === 'forja-accuracy-line') return false;
      for (const [key, ids] of Object.entries(TOGGLES)) if (ids.includes(id)) return !!state.layers[key];
      return true;
    }
    function applyVisibility() {
      if (!state.ready) return;
      for (const layer of map.getStyle().layers || []) if (layer.id.startsWith('forja-')) try { map.setLayoutProperty(layer.id, 'visibility', visible(layer.id) ? 'visible' : 'none'); } catch { /* strat dispărut */ }
    }
    function setBuildings() {
      const l = PALETTES[state.theme] || PALETTES.night;
      try {
        if (map.getLayer('building-3d')) { map.setLayoutProperty('building-3d', 'visibility', state.threeD ? 'visible' : 'none'); map.setPaintProperty('building-3d', 'fill-extrusion-opacity', state.threeD ? l.building3dOpacity : 0); }
        if (map.getLayer('building')) map.setPaintProperty('building', 'fill-opacity', state.threeD ? 0.35 : 1);
      } catch { /* stil fără clădiri */ }
    }
    /** Pulsul lui „Tu” (ca în aplicație); static sub mișcare redusă. */
    function startPulse() {
      if (reduced || state.pulse || typeof root.requestAnimationFrame !== 'function') return;
      let last = 0;
      const frame = t => {
        if (state.destroyed) return;
        state.pulse = root.requestAnimationFrame(frame);
        if (t - last < 50 || (typeof document !== 'undefined' && document.hidden)) return;
        last = t;
        const k = (t % 2000) / 2000;
        try { map.setPaintProperty('forja-me-pulse', 'circle-radius', 14 + k * 22); map.setPaintProperty('forja-me-pulse', 'circle-opacity', 0.32 * (1 - k)); } catch { /* stil în schimbare */ }
      };
      state.pulse = root.requestAnimationFrame(frame);
    }
    function fitTo(coords, maxZoom = 15.5, padding = 64) {
      const b = geo.bounds(coords); if (!b) return false;
      const pad = typeof padding === 'number' ? padding : padding;
      if (b[0][0] === b[1][0] && b[0][1] === b[1][1]) map.jumpTo({center: b[0], zoom: Math.min(maxZoom, 15.5)});
      else map.fitBounds(b, {padding: pad, maxZoom, duration: duration(600)});
      return true;
    }

    Object.assign(api, {
      /** Orice subset: {cells, routes, places, recommended, friends, family, me, devices, focusRoute, pin, selected}. */
      setData(partial) {
        if (!partial) return;
        const kinds = [];
        for (const [k, v] of Object.entries(partial)) { if (v === undefined) continue; state.raw[k] = v; kinds.push(k); }
        try { container.dataset.idle = '0'; } catch { /* container detașat */ }
        render(kinds);
      },
      setMode(mode) { state.mode = mode === 'gasire' ? 'gasire' : 'teren'; applyVisibility(); },
      setLayers(layers) { Object.assign(state.layers, layers || {}); applyVisibility(); return {...state.layers}; },
      get layers() { return {...state.layers}; },
      setTheme(theme) { state.theme = theme === 'day' ? 'day' : 'night'; if (state.ready) { applyPalette(map, state.theme); setBuildings(); } return state.theme; },
      get theme() { return state.theme; },
      set3D(on) {
        if (!state.ready) return false;
        state.threeD = !!on; setBuildings();
        map.easeTo({pitch: state.threeD ? 55 : 0, bearing: state.threeD ? -17 : 0, duration: duration(700)});
        return true;
      },
      /** Pe datele tale: celule, locuri, trasee; altfel poziția ta; altfel camarazii. */
      fitOwn(padding = 64) {
        if (!state.ready) return false;
        const own = [...geo.coordsOf(state.fc.cells), ...geo.coordsOf(state.fc.places), ...geo.coordsOf(state.fc.streets)];
        if (own.length) return fitTo(own, 15.5, padding);
        if (state.me) { map.jumpTo({center: state.me, zoom: 14.5}); return true; }
        const friends = geo.coordsOf(state.fc.friends);
        if (friends.length) return fitTo(friends, 14.5, padding);
        return false;
      },
      fitDevices(padding = 80) { if (!state.ready) return false; const pts = geo.coordsOf(state.fc.devices); if (!pts.length) return false; return fitTo(pts, 16.5, padding); },
      hasOwnData() { return !!(state.fc.cells?.features.length || state.fc.places?.features.length || state.fc.streets?.features.length || state.me); },
      focus(point, zoom = 16, padding) { if (!state.ready || !point) return false; const c = pointOf(point.lat, isNum(point.lng) ? point.lng : point.lon); if (!c) return false; map.easeTo({center: c, zoom: Math.max(map.getZoom(), zoom), duration: duration(700), ...(padding ? {padding} : {})}); return true; },
      fitLine(polyline, padding = 72) { if (!state.ready) return false; const pts = geo.parsePolyline(polyline).map(([lat, lng]) => [lng, lat]); return pts.length ? fitTo(pts, 16, padding) : false; },
      resize() { try { map.resize(); } catch { /* container ascuns */ } },
      destroy() { state.destroyed = true; clearTimeout(timeout); if (state.pulse && root.cancelAnimationFrame) root.cancelAnimationFrame(state.pulse); try { map.remove(); } catch { /* deja eliminată */ } }
    });
    return api;
  }

  root.ForjaMapRenderer = {create, geo, applyPalette, PALETTES, STYLE_URL, supported: () => supported(root.maplibregl)};
})(typeof window !== 'undefined' ? window : globalThis);
