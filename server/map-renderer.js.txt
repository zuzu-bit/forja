/* Shared real geographic renderer: website + Android WebView. No tokens or API access. */
(function(global){
 'use strict';
 const empty=()=>({type:'FeatureCollection',features:[]});
 const collection=value=>value?.type==='FeatureCollection'&&Array.isArray(value.features)?value:empty();
 const feature=(kind,id,p,name)=>({type:'Feature',id:kind+':'+id,geometry:{type:'Point',coordinates:[p.lon,p.lat]},properties:{id:String(id),kind,name:name||'',label:(name||'').trim().split(/\s+/).slice(0,2).map(v=>v[0]||'').join('').toUpperCase()}});
 function create(container,options={}){
  if(!global.maplibregl?.Map||(typeof global.maplibregl.supported==='function'&&!global.maplibregl.supported()))throw Error('Randarea vectorială nu este disponibilă pe acest dispozitiv.');
  let loaded=false,disposed=false,mode3d=false,data={},failed=false;
  const motion=()=>!global.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
  const map=new global.maplibregl.Map({container,style:'https://tiles.openfreemap.org/styles/liberty',center:options.center||[24.9,45.8],zoom:options.zoom||6,attributionControl:true,pitch:0,maxPitch:65,canvasContextAttributes:{antialias:true}});
  map.addControl(new global.maplibregl.NavigationControl({showCompass:true}),'bottom-right');
  const fail=()=>{if(!loaded&&!disposed&&!failed){failed=true;options.onError?.('Harta vectorială nu s-a încărcat. Harta 2D rămâne disponibilă.');}};
  const timeout=setTimeout(fail,20000);
  function paint(){if(!loaded)return;for(const key of ['routes','zones','visits','people','places','devices'])map.getSource('forja-'+key)?.setData(collection(data[key]));}
  map.on('load',()=>{
   if(disposed||failed)return;clearTimeout(timeout);
   // Heights are geographic attributes from OpenMapTiles/OSM, not random blocks.
   const before=map.getStyle().layers.find(l=>l.type==='symbol')?.id;
   map.addSource('forja-buildings',{type:'vector',url:'https://tiles.openfreemap.org/planet',attribution:'<a href="https://openfreemap.org">OpenFreeMap</a> · <a href="https://www.openmaptiles.org">© OpenMapTiles</a> · <a href="https://www.openstreetmap.org/copyright">© OpenStreetMap contributors</a>'});
   map.addLayer({id:'forja-buildings-3d',type:'fill-extrusion',source:'forja-buildings','source-layer':'building',minzoom:14,filter:['all',['!=',['get','hide_3d'],true],['>', ['to-number',['get','render_height'],0],0]],layout:{visibility:'none'},paint:{'fill-extrusion-color':'#c5ccb9','fill-extrusion-height':['to-number',['get','render_height'],0],'fill-extrusion-base':['to-number',['get','render_min_height'],0],'fill-extrusion-opacity':.85}},before);
   for(const key of ['routes','zones','visits','people','places','devices'])map.addSource('forja-'+key,{type:'geojson',data:empty()});
   map.addLayer({id:'forja-zones',type:'fill',source:'forja-zones',paint:{'fill-color':'#71C99B','fill-opacity':.26}});
   map.addLayer({id:'forja-zone-edge',type:'line',source:'forja-zones',paint:{'line-color':'#37875d','line-width':1,'line-opacity':.55}});
   map.addLayer({id:'forja-routes',type:'line',source:'forja-routes',layout:{'line-cap':'round','line-join':'round'},paint:{'line-color':'#4F92FF','line-width':4,'line-opacity':.92}});
   for(const[key,color,radius]of [['visits','#F3B952',9],['places','#f6f6ed',7],['people','#c9e89c',18],['devices','#cbd2d8',10]])map.addLayer({id:'forja-'+key,type:'circle',source:'forja-'+key,paint:{'circle-radius':radius,'circle-color':color,'circle-stroke-color':'#23392b','circle-stroke-width':2}});
   map.addLayer({id:'forja-names',type:'symbol',source:'forja-people',layout:{'text-field':['get','label'],'text-size':12,'text-font':['Noto Sans Regular'],'text-allow-overlap':true},paint:{'text-color':'#20352a'}});
   map.on('click',event=>{const f=map.queryRenderedFeatures(event.point,{layers:['forja-people','forja-visits','forja-places','forja-devices']})[0];if(f){const p=f.properties;options.onPick?.({kind:p.kind,id:p.id,name:p.name||'',lon:f.geometry.coordinates[0],lat:f.geometry.coordinates[1]});}});
   map.on('mousemove',event=>{map.getCanvas().style.cursor=map.queryRenderedFeatures(event.point,{layers:['forja-people','forja-visits','forja-places','forja-devices']}).length?'pointer':'';});
   loaded=true;paint();options.onReady?.();
  });
  map.on('error',()=>{if(!map.isStyleLoaded())fail();});
  map.on('moveend',()=>{const p=map.getCenter();options.onMove?.({lat:p.lat,lon:p.lng,zoom:map.getZoom()});});
  return {
   setData(value){data=value||{};paint();},
   set3D(enabled){if(!loaded)return false;mode3d=!!enabled;map.setLayoutProperty('forja-buildings-3d','visibility',mode3d?'visible':'none');map.easeTo({pitch:mode3d?55:0,duration:motion()?600:0});return true;},
   focus(p,zoom=16){map.flyTo({center:[p.lon,p.lat],zoom,duration:motion()?650:0});},
   fit(points){if(!points.length)return;const bounds=new global.maplibregl.LngLatBounds();points.forEach(p=>bounds.extend([p.lon,p.lat]));map.fitBounds(bounds,{padding:70,maxZoom:15,duration:motion()?600:0});},
   getCenter(){const p=map.getCenter();return {lat:p.lat,lon:p.lng,lng:p.lng};},
   resize(){map.resize();},
   destroy(){disposed=true;clearTimeout(timeout);map.remove();},
   get ready(){return loaded&&!failed;},get mode3d(){return mode3d;}
  };
 }
 global.ForjaMapRenderer={create,feature,collection,empty};
})(typeof window!=='undefined'?window:globalThis);
