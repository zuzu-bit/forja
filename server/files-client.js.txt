const vault={rows:new Map(),cursor:null,serial:0,thumbs:new Map(),preview:null,loading:false,offset:0,moves:new Map()};
const vaultNow=()=>Date.now()+vault.offset;
function vaultURL(blob){return URL.createObjectURL(blob);}
function closeFilePreview(){if(vault.preview?.renderTask)vault.preview.renderTask.cancel();if(vault.preview?.pdfTask)vault.preview.pdfTask.destroy().catch(()=>{});vault.preview?.urls.forEach(URL.revokeObjectURL);vault.preview=null;$('vault-preview').replaceChildren();$('vault-title').textContent='';$('vault-preview-info').textContent='';if($('vault-dialog').open)$('vault-dialog').close();}
function clearFiles(){vaultObserver.disconnect();vault.serial++;vault.loading=false;closeFilePreview();vault.thumbs.forEach(URL.revokeObjectURL);vault.thumbs.clear();vault.rows.clear();vault.moves.clear();vault.cursor=null;$('vault-grid').replaceChildren();$('vault-devices').replaceChildren();$('vault-count').textContent='';$('vault-search').value='';$('vault-kind').value='all';$('vault-more').hidden=true;}
function removeVaultFile(id){const u=vault.thumbs.get(id);if(u)URL.revokeObjectURL(u);vault.thumbs.delete(id);vault.rows.delete(id);$('vault-grid').querySelector('[data-file="'+id+'"]')?.remove();if(vault.preview?.id===id)closeFilePreview();}
async function refreshFileDevices(){
  const current=auth;const data=await api('/v2/files/settings');if(auth!==current)return;
  $('vault-devices').replaceChildren();
  for(const device of data.devices){const row=el('div',undefined,'vault-device');row.append(el('span','Telefon '+device.id.slice(0,8)+' · '+[device.photos?'poze':null,device.files?'fișiere':null].filter(Boolean).join(' + ')+' · '+(device.enabled?'activat':'oprit')));
    if(device.photos||device.files)row.append(button(device.enabled?'Oprește':'Reia sincronizarea',async()=>{await api('/v2/files/settings/'+device.id,{method:'POST',body:JSON.stringify({enabled:!device.enabled,photos:device.photos,files:device.files})});await refreshFileDevices();}));
    $('vault-devices').append(row);
  }
}
async function refreshFiles(more=false,liveUpdate=false){
  if(!auth)return;if(more&&vault.loading)return;
  const current=auth,serial=++vault.serial;vault.loading=true;$('vault-refresh').disabled=true;$('vault-more').disabled=true;
  try{
    const params=new URLSearchParams({kind:$('vault-kind').value,q:$('vault-search').value});if(more&&vault.cursor)params.set('after',vault.cursor);
    const data=await api('/v2/files?'+params);if(auth!==current||serial!==vault.serial)return;
    vault.offset=data.server_at-Date.now();
    if(!more&&!liveUpdate){vaultObserver.disconnect();vault.thumbs.forEach(URL.revokeObjectURL);vault.thumbs.clear();vault.rows.clear();$('vault-grid').replaceChildren();}
    for(const item of liveUpdate?[...data.items].reverse():data.items){const prior=vault.rows.get(item.id);if(prior){for(const key of ['pending_folder','sync_state'])if(!(key in item))delete prior[key];Object.assign(prior,item);updateVaultCard($('vault-grid').querySelector('[data-file="'+item.id+'"]'),prior);continue;}vault.rows.set(item.id,item);if(liveUpdate)$('vault-grid').prepend(vaultCard(item));else $('vault-grid').append(vaultCard(item));}
    if(vault.rows.size)$('vault-grid').querySelectorAll('.empty').forEach(n=>n.remove());
    if(!liveUpdate)vault.cursor=data.next_cursor;$('vault-more').hidden=!vault.cursor;
    $('vault-count').textContent=data.total+' copii în selecție · '+size(data.bytes)+' / 512 MB în cont · se actualizează automat';
    if(!vault.rows.size)empty($('vault-grid'),'Nicio copie disponibilă încă.','După activarea din telefon, noile poze și fișiere apar aici. Copiile expirate sunt eliminate automat.');
    if(!more)await refreshFileDevices();
  }catch(e){if(e.name!=='AbortError'&&serial===vault.serial)notice(e.message);}
  finally{if(serial===vault.serial){vault.loading=false;$('vault-refresh').disabled=false;$('vault-more').disabled=false;}}
}
function vaultCard(item){
  const card=el('article',undefined,'vault-card');card.dataset.file=item.id;
  const cover=el('div',undefined,'vault-cover');cover.append(el('span',item.preview==='image'?'▧':item.preview==='pdf'?'PDF':item.preview==='docx'?'DOCX':'▤'));card.append(cover);
  if(item.thumbnail){const img=el('img');img.alt=item.name;img.loading='lazy';cover.replaceChildren(img);vaultObserver.observe(img);img.dataset.id=item.id;}
  const copy=el('div',undefined,'vault-copy');copy.append(el('h3',item.name),el('p',size(item.bytes)+' · '+dt(item.received_at)),el('p',item.folder||'Fără dosar online','vault-folder'));
  const pending=el('p',undefined,'vault-sync small muted');copy.append(pending);const expiry=el('p',undefined,'vault-expiry');expiry.dataset.expiry=item.expires_at;expiry.textContent=vaultExpiry(item.expires_at);copy.append(expiry);
  const actions=el('div',undefined,'vault-actions');
  if(item.preview!=='download')actions.append(button('Previzualizează',()=>previewFile(item)));
  const move=el('button',item.organizer_job?'Mută pe telefon':'Dosar online','small');move.type='button';move.dataset.vaultMove='yes';move.onclick=async()=>{move.disabled=true;try{await moveVaultFile(item);}catch(e){if(e.name!=='AbortError')notice(e.message);}finally{if(card.isConnected)updateVaultCard(card,item);}};actions.append(button('Descarcă',()=>downloadFile(item)),move,button('Șterge copia',async()=>{if(!confirm('Ștergi copia online a fișierului „'+item.name+'”? Originalul rămâne pe telefon.'))return;await api('/v2/files/'+item.id,{method:'DELETE'});removeVaultFile(item.id);await refreshFiles();}));
  copy.append(actions);card.append(copy);updateVaultCard(card,item);return card;
}
function updateVaultCard(card,item){
 if(!card)return;const move=card.querySelector('[data-vault-move]');if(move){move.disabled=['applied','copied_pending_removal'].includes(item.sync_state);move.textContent=item.sync_state==='applied'?'Mutat pe telefon':item.sync_state==='copied_pending_removal'?'Se verifică mutarea':item.organizer_job?'Mută pe telefon':'Dosar online';}card.querySelector('.vault-folder').textContent=item.folder||'Fără dosar online';const pending=card.querySelector('.vault-sync');pending.hidden=!item.pending_folder;pending.textContent=item.pending_folder?'Mutare cerută → '+item.pending_folder+' · '+({copied_pending_removal:'originalul așteaptă finalizarea',needs_review:'necesită verificare',failed_retryable:'mutarea trebuie reluată'}[item.sync_state]||'așteaptă telefonul'):'';const expiry=card.querySelector('[data-expiry]');expiry.dataset.expiry=item.expires_at;expiry.textContent=vaultExpiry(item.expires_at);
}
async function moveVaultFile(item){
 const current=auth,phoneMove=!!item.organizer_job;
 const folder=prompt(phoneMove?'Mută originalul pe telefon în dosarul:':'Dosar pentru copia online (nu mută originalul din telefon):',item.pending_folder||item.folder||'');if(folder===null)return;
 if(folder.length>(phoneMove?200:120))throw Error('Numele dosarului este prea lung.');
 let request=phoneMove?vault.moves.get(item.id):null;
 if(phoneMove&&request?.folder!==folder){
  if(!confirm('Muți originalul „'+item.name+'” pe telefon în „'+folder+'”? Telefonul poate cere și confirmarea Android.'))return;
  const job=await api('/v2/organizer/devices/'+encodeURIComponent(item.device_id)+'/jobs/'+encodeURIComponent(item.organizer_job));if(auth!==current||!vault.rows.has(item.id))return;
  request={folder,request_id:crypto.randomUUID(),revision:job.revision};vault.moves.set(item.id,request);
 }
 try{const updated=await api('/v2/files/'+item.id,{method:'PATCH',body:JSON.stringify(phoneMove?request:{folder})});if(auth!==current||!vault.rows.has(item.id))return;vault.moves.delete(item.id);Object.assign(item,updated);updateVaultCard($('vault-grid').querySelector('[data-file="'+item.id+'"]'),item);}
 catch(error){if(auth===current&&error.status>=400&&error.status<500)vault.moves.delete(item.id);throw error;}
}
function vaultExpiry(at){const mins=Math.max(0,Math.ceil((at-vaultNow())/60000));return mins>60?'Expiră în '+Math.floor(mins/60)+' h '+mins%60+' min':'Expiră în '+mins+' min';}
const vaultObserver=new IntersectionObserver(entries=>{for(const e of entries)if(e.isIntersecting){vaultObserver.unobserve(e.target);loadVaultThumbnail(e.target);}}, {rootMargin:'200px'});
async function loadVaultThumbnail(img){const id=img.dataset.id,current=auth;try{const blob=await api('/v2/files/'+id+'/thumbnail',{},true);if(auth!==current||!img.isConnected||!vault.rows.has(id)||vault.rows.get(id).expires_at<=vaultNow())return;const url=vaultURL(new Blob([blob],{type:'image/jpeg'}));vault.thumbs.set(id,url);img.src=url;}catch(e){if(e.name!=='AbortError'&&img.isConnected)img.replaceWith(el('span','▧'));}}
async function downloadFile(item){
  const current=auth;const blob=await api('/v2/files/'+item.id,{},true);if(auth!==current||item.expires_at<=vaultNow())throw Error('Copia a expirat.');
  const url=vaultURL(blob),a=el('a');a.href=url;a.download=item.name;document.body.append(a);a.click();a.remove();const timer=setTimeout(()=>{URL.revokeObjectURL(url);urls=urls.filter(u=>u!==url);},30000);urls.push(url);
}
async function previewFile(item){
  closeFilePreview();const current=auth,preview={id:item.id,expires:item.expires_at,urls:[]};vault.preview=preview;
  $('vault-title').textContent=item.name;$('vault-preview-info').textContent='Copia expiră la '+dt(item.expires_at)+'.';$('vault-preview').append(el('p','Se încarcă…','muted'));$('vault-dialog').showModal();
  try{
    const blob=await api('/v2/files/'+item.id,{},true);if(auth!==current||vault.preview!==preview||item.expires_at<=vaultNow())return;
    const body=$('vault-preview');body.replaceChildren();
    if(item.preview==='image'&&['image/jpeg','image/png','image/webp','image/gif','image/avif','image/bmp'].includes(item.media_type)){
      const url=vaultURL(new Blob([blob],{type:item.media_type}));preview.urls.push(url);const img=el('img');img.alt=item.name;img.src=url;body.append(img);
    }else if(item.preview==='pdf'&&item.media_type==='application/pdf'){
      const pdfjs=await import('/insights/pdf.mjs');if(vault.preview!==preview||auth!==current)return;
      pdfjs.GlobalWorkerOptions.workerSrc='/insights/pdf.worker.mjs';
      const bytes=new Uint8Array(await blob.arrayBuffer());if(vault.preview!==preview||auth!==current)return;
      const task=pdfjs.getDocument({data:bytes,isEvalSupported:false,enableXfa:false,useSystemFonts:true,isOffscreenCanvasSupported:false});preview.pdfTask=task;
      const pdf=await task.promise;if(vault.preview!==preview||auth!==current)return;
      const nav=el('div',undefined,'row'),label=el('span',undefined,'muted small'),canvas=el('canvas');canvas.setAttribute('aria-label','Pagina documentului PDF');canvas.style.maxWidth='100%';canvas.style.height='auto';let number=1,busy=false;
      async function renderPDF(){if(busy||vault.preview!==preview)return;busy=true;previous.disabled=next.disabled=true;try{
        const page=await pdf.getPage(number);if(vault.preview!==preview)return;
        const native=page.getViewport({scale:1}),scale=Math.min(1.8,1200/native.width,1600/native.height),viewport=page.getViewport({scale});canvas.width=Math.ceil(viewport.width);canvas.height=Math.ceil(viewport.height);
        preview.renderTask=page.render({canvasContext:canvas.getContext('2d'),viewport});await preview.renderTask.promise;label.textContent='Pagina '+number+' din '+pdf.numPages;
      }catch(e){if(vault.preview===preview&&e.name!=='RenderingCancelledException')notice('Pagina PDF nu poate fi afișată. Poți descărca documentul.');}finally{busy=false;previous.disabled=number<=1;next.disabled=number>=pdf.numPages;}}
      const previous=el('button','Pagina precedentă'),next=el('button','Pagina următoare');previous.addEventListener('click',()=>{if(number>1&&!busy){number--;renderPDF();}});next.addEventListener('click',()=>{if(number<pdf.numPages&&!busy){number++;renderPDF();}});nav.append(previous,label,next);body.append(nav,canvas);await renderPDF();
    }else if(item.preview==='text'){
      const text=await blob.slice(0,256*1024).text();if(vault.preview!==preview||auth!==current)return;body.append(el('pre',text));if(blob.size>256*1024)body.append(el('p','Previzualizare limitată la primele 256 KB. Descarcă fișierul pentru conținutul complet.','muted small'));
    }else if(item.preview==='docx'){
      const xml=await readDocxXml(await blob.arrayBuffer());if(vault.preview!==preview||auth!==current)return;
      const doc=new DOMParser().parseFromString(xml,'application/xml');if(doc.querySelector('parsererror'))throw Error('Document Word invalid. Poți descărca originalul.');
      const paragraphs=Array.from(doc.getElementsByTagNameNS('*','p')).map(p=>Array.from(p.getElementsByTagNameNS('*','t')).map(t=>t.textContent).join(''));
      body.append(el('p','Previzualizare text Word. Formatarea și imaginile sunt disponibile în documentul descărcat.','muted small'),el('pre',paragraphs.join('\n\n')));
    }else body.append(el('p','Acest format poate fi descărcat și deschis în aplicația potrivită.'));
    if(vault.preview===preview&&auth===current)body.append(button('Descarcă originalul',()=>downloadFile(item),'primary'));
  }catch(e){if(e.name!=='AbortError'&&vault.preview===preview){$('vault-preview').replaceChildren(el('p',e.message,'notice'),button('Descarcă originalul',()=>downloadFile(item)));}}
}
