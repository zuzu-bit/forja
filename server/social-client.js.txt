const socialUI={map:null,layer:null,markers:new Map(),data:null,timer:null,chat:null,loading:false,inflight:null,owner:null,fitted:false};
const socialLabel=v=>({walk:'Plimbare',cycle:'Bicicletă',out:'În oraș',partner:'Cuplu'}[v]||'Activitate');
const socialCall=(path,body,method)=>api('/v2/social/'+path,{method:method||(body?'POST':'GET'),...(body?{body:JSON.stringify(body)}:{})});
const socialFresh=p=>!!p&&p.at>Date.now()-120000;
const socialMotion=()=>!window.matchMedia('(prefers-reduced-motion: reduce)').matches;
const socialAge=at=>{const seconds=Math.max(0,Math.floor((Date.now()-at)/1000));return seconds<10?'acum':seconds<60?'acum '+seconds+' sec':seconds<3600?'acum '+Math.floor(seconds/60)+' min':'la '+dt(at);};
function socialAvatar(name,self=false){const badge=el('span',name.trim().split(/\s+/).slice(0,2).map(s=>s[0]||'').join('').toUpperCase(),'social-avatar'+(self?' social-self-pin':''));badge.setAttribute('aria-hidden','true');return badge;}
function socialMore(key,label='Mai multe'){const details=el('details',undefined,'social-more');details.dataset.socialMenu=key;details.append(el('summary',label));return details;}
function socialEmpty(container,symbol,title){const row=el('div',undefined,'social-empty');const icon=el('span',symbol);icon.setAttribute('aria-hidden','true');row.append(icon,el('p',title));container.append(row);}
function socialShowMap(p,zoom=16){if(typeof journeyFocus==='function')journeyFocus(p,zoom);socialUI.map?.flyTo([p.lat,p.lon],zoom,{duration:.7,animate:socialMotion()});if(window.matchMedia('(max-width: 1000px)').matches)$('social-map')?.scrollIntoView({behavior:socialMotion()?'smooth':'instant',block:'center'});}
function socialFit(){if(!socialUI.map)return;const d=socialUI.data;let points=d?[d.me,...d.friends].filter(f=>socialFresh(f.location)).map(f=>[f.location.lat,f.location.lon]):[];if(!points.length&&typeof recoveryUI!=='undefined')points=(recoveryUI.data||[]).filter(d=>d.position?.at>Date.now()-86400000).map(d=>[d.position.lat,d.position.lon]);if(!points.length)return;socialUI.map.fitBounds(points,{padding:[64,100],maxZoom:15,animate:socialMotion()});socialUI.fitted=true;if(typeof journeyUI!=='undefined')journeyUI.vector?.fit(points.map(p=>({lat:p[0],lon:p[1]})));}
function clearSocial(){if(typeof clearJourney==='function')clearJourney();clearRecovery();clearInterval(socialUI.timer);socialUI.timer=null;socialUI.data=null;socialUI.chat=null;socialUI.owner=null;socialUI.loading=false;socialUI.inflight=null;socialUI.fitted=false;socialUI.markers.clear();if(socialUI.map){socialUI.map.remove();socialUI.map=null;socialUI.layer=null;}for(const id of ['social-friends','social-groups','social-places','social-chat','social-profile','social-partner'])$(id)?.replaceChildren();}
async function refreshSocial(){
 if(socialUI.loading)return socialUI.inflight;if(!auth||$('page-social').hidden||document.hidden)return;const current=auth;socialUI.loading=true;
 socialUI.inflight=(async()=>{try{const d=await socialCall('state');if(auth!==current)return;socialUI.data=d;renderSocial();if(socialUI.chat)await socialChat(socialUI.chat,false);}catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}finally{if(auth===current){socialUI.loading=false;socialUI.inflight=null;}}})();return socialUI.inflight;
}
function openSocial(){if(!socialUI.map){socialUI.map=L.map('social-map',{zoomControl:false,scrollWheelZoom:false}).setView([45.8,24.9],6);L.control.zoom({position:'bottomright'}).addTo(socialUI.map);L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png',{maxZoom:19,attribution:'&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap contributors</a>'}).addTo(socialUI.map);socialUI.layer=L.layerGroup().addTo(socialUI.map);}setTimeout(()=>socialUI.map?.invalidateSize(),0);refreshSocial();refreshRecovery();if(typeof openJourney==='function')openJourney();clearInterval(socialUI.timer);socialUI.timer=setInterval(()=>{expireSocialMarkers();recoveryMarkers();refreshSocial();refreshRecovery();if(typeof refreshJourney==='function'){refreshJourney();journeyFlush();}},15000);}
function expireSocialMarkers(){const d=socialUI.data;if(!d)return;let changed=false;for(const f of [...d.friends,d.me])if(f.location&&!socialFresh(f.location)){f.location=null;changed=true;}if(d.me.session&&!d.me.session.continuous&&d.me.session.until<=Date.now()){d.me.session=null;d.me.location=null;changed=true;}if(changed)renderSocial();else renderSocialMap();}
async function socialAction(path,body,method,success){const current=auth;try{await socialCall(path,body,method);if(auth===current){if(socialUI.loading)await socialUI.inflight;if(auth!==current)return false;await refreshSocial();if(auth!==current)return false;if(success)notice(success,true);return true;}}catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}return false;}
function socialNav(p){const a=el('a','Itinerar ↗');a.href=`https://www.openstreetmap.org/directions?to=${encodeURIComponent(p.lat+','+p.lon)}`;a.target='_blank';a.rel='noopener noreferrer';return a;}
function renderSocialMap(){
 if(!socialUI.map||!socialUI.data)return;const d=socialUI.data,ids=new Set();
 for(const f of [{...d.me,name:'Tu'},...d.friends]){const l=f.location;if(!socialFresh(l))continue;ids.add(f.id);let marker=socialUI.markers.get(f.id);
  if(!marker){const badge=socialAvatar(f.name,f.id===d.me.id);badge.classList.add('social-pin');badge.title=f.name;marker=L.marker([l.lat,l.lon],{title:f.name,alt:f.name,icon:L.divIcon({html:badge,className:'social-marker',iconSize:[48,52],iconAnchor:[24,48]})}).addTo(socialUI.layer);socialUI.markers.set(f.id,marker);}else marker.setLatLng([l.lat,l.lon]);
  const popup=el('div',undefined,'social-map-popup');popup.append(el('strong',f.name),el('p',socialAge(l.at)+' · ±'+Math.round(l.accuracy)+' m','social-meta'),el('p',(l.battery==null?'Baterie indisponibilă':l.battery+'% baterie'),'social-meta'));const actions=el('div',undefined,'social-actions');if(f.id!==d.me.id)actions.append(button('Mesaj',()=>socialChat(f),'social-primary'));actions.append(socialNav(l));popup.append(actions);if(marker.getPopup())marker.setPopupContent(popup);else marker.bindPopup(popup);
 }
 for(const[id,marker]of socialUI.markers)if(!ids.has(id)){socialUI.layer.removeLayer(marker);socialUI.markers.delete(id);}
 const count=d.friends.filter(f=>socialFresh(f.location)).length;if($('social-map-summary'))$('social-map-summary').textContent=count?(count===1?'1 prieten pe hartă':count+' prieteni pe hartă'):'Nicio poziție recentă a prietenilor';if($('social-map-fit'))$('social-map-fit').disabled=!ids.size&&!(typeof recoveryUI!=='undefined'&&recoveryUI.data?.some(d=>d.position?.at>Date.now()-86400000));if(!socialUI.fitted&&ids.size)socialFit();
