const socialUI={map:null,layer:null,data:null,timer:null,chat:null,loading:false,owner:null};
const socialLabel=v=>({walk:'Mers',cycle:'Cycling',out:'În oraș',partner:'Cuplu'}[v]||'Activitate');
const socialCall=(path,body,method)=>api('/v2/social/'+path,{method:method||(body?'POST':'GET'),...(body?{body:JSON.stringify(body)}:{})});
function clearSocial(){clearRecovery();clearInterval(socialUI.timer);socialUI.timer=null;socialUI.data=null;socialUI.chat=null;socialUI.owner=null;socialUI.loading=false;if(socialUI.map){socialUI.map.remove();socialUI.map=null;socialUI.layer=null;}for(const id of ['social-friends','social-groups','social-places','social-chat','social-profile','social-partner'])$(id)?.replaceChildren();}
async function refreshSocial(){
 if(!auth||$('page-social').hidden||document.hidden||socialUI.loading)return;const current=auth;socialUI.loading=true;
 try{const d=await socialCall('state');if(auth!==current)return;socialUI.data=d;renderSocial();if(socialUI.chat)await socialChat(socialUI.chat,false);}catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}finally{if(auth===current)socialUI.loading=false;}
}
function openSocial(){if(!socialUI.map){socialUI.map=L.map('social-map',{zoomControl:true}).setView([45.8,24.9],6);L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png',{maxZoom:19,attribution:'&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap contributors</a>'}).addTo(socialUI.map);socialUI.layer=L.layerGroup().addTo(socialUI.map);}setTimeout(()=>socialUI.map?.invalidateSize(),0);refreshSocial();refreshRecovery();clearInterval(socialUI.timer);socialUI.timer=setInterval(()=>{expireSocialMarkers();recoveryMarkers();refreshSocial();refreshRecovery();},15000);}
function expireSocialMarkers(){const d=socialUI.data;if(!d)return;for(const f of [...d.friends,d.me])if(f.location&&f.location.at<=Date.now()-120000)f.location=null;renderSocialMap();}
async function socialAction(path,body,method){const current=auth;try{await socialCall(path,body,method);if(auth===current)await refreshSocial();}catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}}
function socialNav(p){const a=el('a','Navigare ↗');a.href=`https://www.openstreetmap.org/directions?to=${encodeURIComponent(p.lat+','+p.lon)}`;a.target='_blank';a.rel='noopener noreferrer';return a;}
function renderSocialMap(){if(!socialUI.map||!socialUI.data)return;socialUI.layer.clearLayers();const d=socialUI.data;for(const f of [{...d.me,name:'Tu'},...d.friends]){const l=f.location;if(!l||l.at<=Date.now()-120000)continue;const badge=el('span',f.name.split(/\s+/).slice(0,2).map(s=>s[0]||'').join(''),'social-pin');badge.title=f.name;const marker=L.marker([l.lat,l.lon],{icon:L.divIcon({html:badge,className:'social-marker',iconSize:[48,52],iconAnchor:[24,48]})}).addTo(socialUI.layer);const popup=el('div');popup.append(el('strong',f.name),el('p',`±${Math.round(l.accuracy)} m · ${l.battery}% baterie`));marker.bindPopup(popup);if(f.id!==d.me.id)marker.on('click',()=>socialChat(f));}}
function renderSocial(){const d=socialUI.data;if(!d)return;renderSocialMap();$('social-state').textContent=d.me.session?(d.me.session.continuous?'Cuplu · partajare doar cu partenerul, până o oprești':`${socialLabel(d.me.session.mode)} · partajare până la ${dt(d.me.session.until)}`):'Mod fantomă · locația ta nu este partajată';$('social-ghost').disabled=!d.me.session;
 renderPartner(d);
 const box=$('social-friends');box.replaceChildren();if(!d.friends.length)box.append(el('p','Invită primul prieten. Pozițiile apar numai după acceptare și pornirea partajării.','muted'));
 for(const f of d.friends){const card=el('article',undefined,'social-card'),l=f.location;card.append(el('strong',f.name),el('p',l?`${socialLabel(f.mode)} · ${(l.speed*3.6).toFixed(1)} km/h · ${l.battery}% baterie`:'Poziție indisponibilă / mod fantomă','muted small'));if(l){card.append(button('Arată pe hartă',()=>socialUI.map.flyTo([l.lat,l.lon],16,{duration:.8})),socialNav(l));if(f.checkin)card.append(el('p','Spune că este la: '+f.checkin.name));}card.append(button('Conversație',()=>socialChat(f)),button('Elimină',()=>socialAction('friend',{id:f.id,action:'remove'})),button('Blochează',()=>socialAction('friend',{id:f.id,action:'block'})));box.append(card);}
 for(const f of d.incoming){const row=el('article',undefined,'social-card');row.append(el('strong',f.name+' vrea să fiți prieteni'),button('Acceptă',()=>socialAction('friend',{id:f.id,action:'accept'})),button('Refuză',()=>socialAction('friend',{id:f.id,action:'reject'})));box.append(row);}
 const profile=$('social-profile');if(!profile.childNodes.length){const name=el('input');name.maxLength=40;name.value=d.me.name;name.setAttribute('aria-label','Nume pentru prieteni');const code=el('input');code.readOnly=true;code.value=d.me.code;code.setAttribute('aria-label','Codul tău de invitație');profile.append(el('h3','Profilul și codul tău'),name,button('Salvează numele',()=>socialAction('profile',{name:name.value.trim()})),code,el('p','Trimite acest cod prietenului. El introduce codul, iar tu accepți cererea.','muted small'));}
 const places=$('social-places');places.replaceChildren();for(const p of d.me.places){const row=el('article',undefined,'social-card');row.append(el('strong',p.name),button('Pe hartă',()=>socialUI.map.flyTo([p.lat,p.lon],16)),socialNav(p),button('Elimină',()=>socialAction('place',{id:p.id},'DELETE')));places.append(row);}if(!d.me.places.length)places.append(el('p','Salvează locuri private din centrul hărții.','muted'));
 const groups=$('social-groups');groups.replaceChildren();for(const g of d.groups){const row=el('article',undefined,'social-card');row.append(el('strong',g.name),el('p',`${socialLabel(g.mode)} · ${dt(g.at)} · ${g.place}`),el('p',`${g.going.length}/${g.members.length} participă`),button(g.going.includes(d.me.id)?'Nu mai vin':'Vin și eu',()=>socialAction('group-response',{id:g.id,going:!g.going.includes(d.me.id)})),socialNav(g),button('Ieși din grup',()=>socialAction('leave-group',{id:g.id})));groups.append(row);}if(!d.groups.length)groups.append(el('p','Creează o plimbare, o tură pe bicicletă sau o întâlnire.','muted'));
 const members=$('social-group-members');const selected=new Set([...members.querySelectorAll('input:checked')].map(e=>e.value));members.replaceChildren();for(const f of d.friends){const label=el('label'),input=el('input');input.type='checkbox';input.value=f.id;input.checked=selected.has(f.id);label.append(input,document.createTextNode(f.name));members.append(label);}
 $('social-explored').textContent=`${d.me.explored.length} zone explorate · vizibile doar pentru tine`;
 if(socialUI.chat&&!d.friends.some(f=>f.id===socialUI.chat.id)){$('social-chat').replaceChildren();socialUI.chat=null;}
}
async function socialChat(f,build=true){const current=auth;socialUI.chat=f;const box=$('social-chat');if(build){box.replaceChildren();box.append(el('h3','Conversație · '+f.name),el('p','Ultimele 100 de mesaje, maximum 7 zile.','muted small'));const messages=el('div');messages.id='social-messages';const text=el('textarea');text.maxLength=1000;text.setAttribute('aria-label','Mesaj');const send=button('Trimite',async()=>{if(!text.value.trim())return;send.disabled=true;const draft=text.value;try{await socialCall('chat',{friend:f.id,id:crypto.randomUUID(),text:draft});if(auth!==current||socialUI.chat?.id!==f.id)return;text.value='';await socialChat(f,false);}catch(e){if(auth===current)notice(e.message);}finally{send.disabled=false;}});box.append(messages,text,send);box.scrollIntoView({behavior:'smooth',block:'nearest'});}try{const result=await socialCall('chat?friend='+encodeURIComponent(f.id));if(auth!==current||socialUI.chat?.id!==f.id)return;const messages=$('social-messages');messages.replaceChildren();for(const m of result.messages)messages.append(el('p',(m.from===socialUI.data.me.id?'Tu: ':f.name+': ')+m.text));}catch(e){if(auth===current)notice(e.message);}}
$('social-ghost').onclick=()=>socialAction('session',null,'DELETE');
$('social-invite-form').onsubmit=async e=>{e.preventDefault();await socialAction('invite',{code:$('social-invite').value.trim()});};
$('social-place-form').onsubmit=async e=>{e.preventDefault();const p=socialUI.map.getCenter();await socialAction('place',{name:$('social-place-name').value.trim(),lat:p.lat,lon:p.lng});};
$('social-group-form').onsubmit=async e=>{e.preventDefault();const p=socialUI.map.getCenter();await socialAction('group',{name:$('social-group-name').value.trim(),mode:$('social-group-mode').value,at:new Date($('social-group-at').value).getTime(),place:$('social-group-place').value.trim(),lat:p.lat,lon:p.lng,friends:[...$('social-group-members').querySelectorAll('input:checked')].map(e=>e.value)});};

function renderPartner(d){
 const box=$('social-partner');box.replaceChildren();const p=d.me.partner;
 box.append(el('h2','Cuplu · mereu aproape'),el('p','Fiecare pornește partajarea continuă pe propriul telefon, din Hartă → Cuplu & contacte. Acceptarea asocierii nu pornește GPS-ul.','muted small'));
 if(!p){if(!d.friends.length)box.append(el('p','Adaugă întâi partenerul ca prieten.'));for(const f of d.friends)box.append(button('Invită în cuplu · '+f.name,()=>socialAction('partner',{id:f.id,action:'invite'})));}
 else{box.append(el('strong',p.name));if(p.state==='invited')box.append(el('p','Ai o invitație de cuplu.'),button('Acceptă asocierea',()=>socialAction('partner',{id:p.id,action:'accept'})));else box.append(el('p',p.state==='accepted'?'Asociere acceptată. Fiecare își controlează propria partajare.':'Așteaptă acceptarea partenerului.'));box.append(button(p.state==='accepted'?'Închide asocierea de cuplu':'Anulează / refuză',()=>socialAction('partner',{id:p.id,action:p.state==='accepted'?'disconnect':'decline'})));}
 box.append(el('p','Ultima poziție expiră după două minute fără actualizări. Oprirea din Mod fantomă revocă sesiunea; telefonul nu o poate relua automat după revocare.','muted small'));
 box.append(el('h3','Prieteni din contacte'),el('p','Accesul la agendă, verificarea numărului și sincronizarea se configurează pe telefon. Numele din agenda ta rămân pe telefon.','muted small'));
 if(d.me.discoverable)box.append(el('p','Ai activat găsirea după numărul verificat.'),button('Oprește găsirea după numărul meu',()=>socialAction('contacts/discovery',null,'DELETE')));
}
