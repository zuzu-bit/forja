function h(tag, props, ...kids) {
  const n = document.createElement(tag);
  if (props) for (const [k, v] of Object.entries(props)) {
    if (v === null || v === undefined || v === false) continue;
    if (k === 'class') n.className = v;
    else if (k === 'text') n.textContent = v;
    else if (k === 'style' && typeof v === 'object') Object.assign(n.style, v);
    else if (k === 'dataset') Object.assign(n.dataset, v);
    else if (k.startsWith('on') && typeof v === 'function') n.addEventListener(k.slice(2), v);
    else n.setAttribute(k, v === true ? '' : String(v));
  }
  append(n, kids);
  return n;
}
function append(n, kids) {
  for (const c of [kids].flat(Infinity)) { if (c === null || c === undefined || c === false) continue; n.append(c instanceof Node ? c : String(c)); }
  return n;
}
const SVGNS = 'http://www.w3.org/2000/svg';

  const n = document.createElementNS(SVGNS, tag);
  if (attrs) for (const [k, v] of Object.entries(attrs)) if (v !== null && v !== undefined && v !== false) n.setAttribute(k, String(v));
  for (const c of kids.flat(Infinity)) if (c) n.append(c);
  return n;
}

/** Plural românesc: 1 zonă · 2 zone · 20 de zone. */
function plural(n, one, many) {
  const v = Math.abs(Math.round(n)), r = v % 100;
  if (v === 1) return fmt.int(n) + ' ' + one;
  if (v === 0 || (r >= 1 && r <= 19)) return fmt.int(n) + ' ' + many;
  return fmt.int(n) + ' de ' + many;
}

function distanceM(a, b) {
  if (!a || !b || [a.lat, a.lng, b.lat, b.lng].some(v => typeof v !== 'number')) return null;
  const R = 6371000, rad = Math.PI / 180, dLat = (b.lat - a.lat) * rad, dLng = (b.lng - a.lng) * rad;
  const x = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * rad) * Math.cos(b.lat * rad) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.asin(Math.min(1, Math.sqrt(x)));
}

/** Cerere autentificată spre același server. JSON implicit; `blob:true` întoarce Blob. Un 401 reîmprospătează tokenul o dată. */
async function api(path, opts = {}) {
  const current = Auth.session;
  if (!current) throw new ApiError('Intră din nou în cont.', 401);
  const send = async token => fetch(path, {method: opts.method || 'GET', headers: {'Authorization': 'Bearer ' + token, ...(opts.body !== undefined ? {'content-type': 'application/json'} : {}), ...(opts.headers || {})},
    body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined, signal: opts.signal || current.controller.signal, cache: 'no-store'});
  let r;
  try { r = await send(await Auth.token()); if (r.status === 401 && Auth.session === current) r = await send(await Auth.renew(current)); }
  catch (e) {
    if (e instanceof ApiError && e.status === 401) { Site.expired(); throw new DOMException('Sesiune expirată', 'AbortError'); }
    if (e.name === 'AbortError' || e instanceof ApiError) throw e;
    throw new ApiError('Fără legătură. Verifică internetul.', 0);
  }
  if (Auth.session !== current) throw new DOMException('Cont schimbat', 'AbortError');
  if (r.status === 401) { Site.expired(); throw new DOMException('Sesiune expirată', 'AbortError'); }
  if (!r.ok) { let data = null; try { data = await r.json(); } catch { /* fără corp */ } throw new ApiError(cleanError(data?.error, r.status), r.status); }
  if (opts.blob) return r.blob();
  if (r.status === 204) return null;
  return r.json();
}
/** Mesajele serverului sunt în română; orice altceva (engleză, HTTP brut) devine un mesaj uman. */
function cleanError(message, status) {
  const english = /\b(not|found|invalid|error|server|failed|file|unauthori[sz]ed|forbidden|missing|the|is|of|http)\b/i;
  if (typeof message === 'string' && message.trim() && message.length <= 200 && (/[ăâîșț]/i.test(message) || !english.test(message))) return message.trim();
  if (status === 404) return 'Nu mai este disponibil.';
  if (status === 429) return 'Prea multe cereri. Mai încearcă peste un minut.';
  return 'Nu a mers. Încearcă din nou.';
}

// ───────────────────────── Sondare doar cât e nevoie ─────────────────────────
const Poll = {
  jobs: new Map(),
  /** Rulează `fn` la fiecare `ms` cât pagina e vizibilă; `immediate` o rulează și acum. */
  every(key, ms, fn, immediate = false) {
    Poll.stop(key);
    const job = {ms, fn, timer: null, busy: false};
    Poll.jobs.set(key, job);
    const tick = async () => {
      if (Poll.jobs.get(key) !== job) return;
      if (document.visibilityState === 'visible' && !job.busy) { job.busy = true; try { await job.fn(); } catch { /* fiecare secțiune își arată eroarea */ } finally { job.busy = false; } }
      if (Poll.jobs.get(key) === job) job.timer = setTimeout(tick, job.ms);
    };
    job.kick = () => { clearTimeout(job.timer); tick(); };
    if (immediate) tick(); else job.timer = setTimeout(tick, ms);
    return job;
  },
  stop(key) { const j = Poll.jobs.get(key); if (j) { clearTimeout(j.timer); Poll.jobs.delete(key); } },
  stopAll() { for (const key of [...Poll.jobs.keys()]) Poll.stop(key); },
  /** La revenirea în tab, toate sondările active rulează imediat. */
  wake() { for (const j of Poll.jobs.values()) j.kick?.(); }
};
document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'visible') Poll.wake(); });

/** Încarcă o secțiune: schelet la prima încărcare, eroare cu „Reîncearcă”, apoi render. */
async function loadInto(id, body, fetcher, render, opts = {}) {
  const first = !body.dataset.loaded;
  if (first) fill(body, skeleton(opts.skeleton));
  try {
    const data = await fetcher();
    render(data);
    body.dataset.loaded = '1';
    return data;
  } catch (e) {
    if (e?.name === 'AbortError') return null;
    if (first || opts.replaceOnError) fill(body, errorState(e?.message, () => loadInto(id, body, fetcher, render, opts)));
    else toast(e?.message || 'Actualizarea nu a reușit.');
    return null;
  }
}

/** `#sectiune[/detaliu]`; detaliul vine codificat din aplicație (Uri.encode). */
function parseHash(hash = location.hash) {
  const raw = String(hash || '').replace(/^#/, '');
  const slash = raw.indexOf('/');
  const id = (slash < 0 ? raw : raw.slice(0, slash)).toLowerCase();
  let detail = slash < 0 ? '' : raw.slice(slash + 1);
  try { detail = decodeURIComponent(detail); } catch { /* rămâne brut */ }
  return {id, detail};
}
function go(id, detail = '') { const target = '#' + id + (detail ? '/' + encodeURIComponent(detail).replace(/%2C/gi, ',') : ''); if (location.hash !== target) location.hash = target; else route(); }
function route() {
  if (!Auth.session) return;
  let {id, detail} = parseHash();
  if (!SECTION[id]) { history.replaceState(null, '', location.pathname + location.search + '#azi'); id = 'azi'; detail = ''; }
  activate(id, detail);
}
function activate(id, detail) {
  const x = SECTION[id], prev = Site.current, changed = prev !== id;
  if (changed && prev) { Site.modules[prev]?.leave?.(); for (const key of [...Poll.jobs.keys()]) if (key.startsWith(prev + ':')) Poll.stop(key); }
  Site.current = id; Site.detail = detail;
  for (const y of SECTIONS) { const el = $('s-' + y.id); if (el) el.hidden = y.id !== id; }
  mountSection(id);
  document.body.dataset.section = id;
  document.body.classList.toggle('night', !!x.night);
  document.body.classList.toggle('full', !!x.full);
  for (const a of document.querySelectorAll('[data-nav]')) { const on = a.dataset.nav === id; a.classList.toggle('active', on); if (on) a.setAttribute('aria-current', 'page'); else a.removeAttribute('aria-current'); }
  $('tab-more')?.classList.toggle('active', !['azi', 'teren', 'gasire'].includes(id));
  document.title = 'FORJA · ' + x.nav;
  if (changed) { window.scrollTo(0, 0); $('t-' + id)?.focus({preventScroll: true}); }
  Site.modules[id]?.enter?.(detail, changed);
}
window.addEventListener('hashchange', route);
