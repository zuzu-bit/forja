async function createOrganizerJob(draft){

 if(draft.scope==='folder'&&!draft.folder.trim())throw Error('Alege albumul sau subdosarul.');
 if(!draft.destination.trim())throw Error('Alege un dosar destinație.');
 if(draft.mode==='online'&&!draft.ai_consent)throw Error('Permite analiza online înainte de pornire.');
 const source=(d.grant.sources||[]).find(row=>row.id===draft.source_id&&row.source===draft.source),relative=draft.scope==='folder'?draft.folder.trim():'';const folder=relative;
 const values={grant_id:d.grant.id,revision:d.revision,source:draft.source,...(source?{source_id:source.id}:{}),scope:{folder,recursive:true},destination:draft.destination.trim(),mode:draft.mode,auto_apply:draft.auto_apply&&d.grant.organize===true,ai_consent:draft.mode==='online'&&draft.ai_consent};
 const key=JSON.stringify(values);if(organizerV4.draftKey?.key!==key)organizerV4.draftKey={key,id:crypto.randomUUID()};

 organizerV4.jobs.set(job.id,job);renderOrganizerJobs();
 await sendOrganizerCommand(job,'continue',draft.count);if(auth!==current)return;organizerV4.draftKey=null;
}

function renderOrganizerJobs(){
 const box=$('organizer-jobs'),device=organizerV4.device;if(!device)return;
 for(const card of box.querySelectorAll('[data-organizer-job]'))if(!organizerV4.jobs.has(card.dataset.organizerJob)){organizerV4.itemsSerial.set(card.dataset.organizerJob,(organizerV4.itemsSerial.get(card.dataset.organizerJob)||0)+1);card.remove();}
 box.querySelectorAll(':scope>.organizer-job-empty').forEach(n=>n.remove());
 if(!organizerV4.jobs.size){box.append(el('p','Lucrările tale apar aici după pornire.','organizer-job-empty'));return;}
 for(const job of [...organizerV4.jobs.values()].sort((a,b)=>b.created_at-a.created_at)){
  let card=[...box.querySelectorAll('[data-organizer-job]')].find(node=>node.dataset.organizerJob===job.id);
  if(!card){card=el('article',undefined,'organizer-job');card.dataset.organizerJob=job.id;card.append(el('div',undefined,'organizer-job-summary'),el('div',undefined,'organizer-job-actions'));const detail=cleanupDetails('Vezi rezultatul'),body=el('div',undefined,'organizer-items');detail.classList.add('organizer-job-detail');detail.append(body);detail.addEventListener('toggle',()=>{if(detail.open&&!body.dataset.loaded)openOrganizerJobItems(job.id,body);});card.append(detail);box.append(card);}
  const summary=card.querySelector('.organizer-job-summary');summary.replaceChildren();const head=el('div',undefined,'organizer-job-head');head.append(el('h3',job.source==='photos'?'Galerie':'Documente'),el('span',organizerJobLabel[job.state]||'Se verifică starea','pill'));summary.append(head,el('p',(job.scope?.folder||'Toată sursa autorizată')+' → '+job.destination,'small muted'));
  const counts=el('div',undefined,'organizer-counts');for(const [key,label]of [['moved','mutate'],['uploaded','copii de organizat'],['needs_review','de verificat'],['failed_retryable','de reluat']]){const n=organizerCount(job,key);if(n!==null){const cell=el('div');cell.append(el('b',String(n)),el('span',label));counts.append(cell);}}summary.append(counts);
  const total=organizerCount(job,'total'),moved=organizerCount(job,'moved');if(total>0&&moved!==null){const progress=el('progress',undefined,'organizer-progress');progress.max=total;progress.value=Math.min(total,moved);progress.setAttribute('aria-label','Fișiere mutate pe telefon');summary.append(progress,el('p',moved+' din '+total+' mutate','small muted'));}
  if(job.message)summary.append(el('p',job.message,'small muted'));
  if(organizerCount(job,'copied_pending_removal'))summary.append(el('p',job.counters.copied_pending_removal+' copii create · originalele încă așteaptă finalizarea mutării.','small muted'));
  const actions=card.querySelector('.organizer-job-actions');actions.replaceChildren();const row=el('div',undefined,'row'),pending=organizerCommandPending(job),busy=organizerV4.busy.has(job.id),d=organizerDevice();
  if(!['cancelled','access_needed'].includes(job.state)){
   if(!pending&&(job.command?.status==='complete'||job.state==='paused'||!job.command||job.state==='complete')){let count=organizerV4.batchCounts.get(job.id)??job.command?.count??50;const choice=el('select');choice.setAttribute('aria-label','Mărimea următorului lot');for(const value of [...new Set([50,100,0,count])])organizerOption(choice,String(value),value?'Următoarele '+value:'Toate cele rămase');choice.value=String(count);choice.onchange=()=>{count=Number(choice.value);organizerV4.batchCounts.set(job.id,count);};if(job.command?.status==='complete'||job.state==='complete')row.append(choice);row.append(button(job.command?.status==='complete'||job.state==='complete'?'Organizează următorul lot':job.command?'Continuă':'Pornește',()=>sendOrganizerCommand(job,'continue',count),'primary'));}
   if(['running','partial','awaiting_phone'].includes(job.state)&&job.command?.status!=='complete')row.append(button('Pauză',()=>sendOrganizerCommand(job,'pause',0)));
   row.append(button('Oprește lucrarea',()=>{if(confirm('Oprești operațiile noi ale acestei lucrări? O mutare începută se verifică pe telefon.'))return sendOrganizerCommand(job,'cancel',0);},'small subtle'));
  }
  row.append(button('Actualizează',async()=>{await refreshOrganizerJobs();const detail=card.querySelector('.organizer-job-detail');if(detail.open)await openOrganizerJobItems(job.id,detail.querySelector('.organizer-items'));}));for(const b of row.querySelectorAll('button'))b.disabled=busy||!d?.grant?.enabled;actions.append(row);
  if(pending)actions.append(el('p','Comanda așteaptă răspunsul telefonului.','organizer-form-status'));
  const details=card.querySelector('.organizer-job-detail'),body=details.querySelector('.organizer-items'),signature=JSON.stringify([job.updated_at,job.state,job.counters]);
  if(details.open&&body.dataset.loaded&&body.dataset.jobSignature!==signature&&!body.contains(document.activeElement))openOrganizerJobItems(job.id,body);
 }
}
async function sendOrganizerCommand(job,action,count){
 const current=auth,device=organizerV4.device;if(!current||!device||organizerV4.busy.has(job.id))return;
 const live=organizerV4.jobs.get(job.id);if(!live)throw Error('Lucrarea nu mai este disponibilă.');
 const fingerprint=JSON.stringify([device,job.id,action,count]);let request=organizerV4.requests.get(job.id);if(request?.fingerprint!==fingerprint){request={fingerprint,request_id:crypto.randomUUID(),revision:live.revision};organizerV4.requests.set(job.id,request);}
 organizerV4.busy.add(job.id);organizerV4.serial++;renderOrganizerJobs();
 try{const next=await api(organizerApi(device,job.id)+'/command',{method:'POST',body:JSON.stringify({request_id:request.request_id,revision:request.revision,action,count})});if(auth!==current||device!==organizerV4.device)return;organizerV4.serial++;organizerV4.jobs.set(next.id,next);organizerV4.requests.delete(job.id);renderOrganizerJobs();}
 catch(error){if(auth===current&&device===organizerV4.device&&error.status>=400&&error.status<500){organizerV4.requests.delete(job.id);await refreshOrganizerJobs();}throw error;}
 finally{if(auth===current&&device===organizerV4.device){organizerV4.busy.delete(job.id);renderOrganizerJobs();}}
}
async function openOrganizerJobItems(jobId,body,after=null){
 const current=auth,device=organizerV4.device,loadedJob=organizerV4.jobs.get(jobId),signature=JSON.stringify([loadedJob?.updated_at,loadedJob?.state,loadedJob?.counters]),serial=(organizerV4.itemsSerial.get(jobId)||0)+1;organizerV4.itemsSerial.set(jobId,serial);
 try{const data=await api(organizerApi(device,jobId)+'/items'+(after?'?after='+encodeURIComponent(after):''));if(auth!==current||device!==organizerV4.device||serial!==organizerV4.itemsSerial.get(jobId)||!body.isConnected)return;
  if(!after){body.replaceChildren();for(const [key,edit]of organizerV4.edits)if(key.startsWith(jobId+':'))edit.visible=false;}body.dataset.loaded='yes';body.dataset.jobSignature=signature;body.querySelector('.organizer-more')?.remove();const job=organizerV4.jobs.get(jobId);
  if(!data.items.length&&!after)body.append(el('p','Telefonul nu a trimis încă rezultate pentru această lucrare.','small muted'));
  for(const item of data.items){const card=el('article',undefined,'organizer-item');card.dataset.organizerItem=item.id;const head=el('div',undefined,'organizer-item-head');head.append(el('strong',item.name),el('span',organizerItemLabel[item.state]||'Se verifică','pill'));card.append(head);if(item.destination)card.append(el('p','→ '+item.destination,'small muted'));if(item.reason)card.append(el('p',item.reason,'small muted'));if(item.partial||item.coverage?.partial)card.append(el('p','Analiză parțială · de verificat','small muted'));if(item.analysis)card.append(organizerEvidence(item.analysis));
   const file=item.received===true?item.file:null;if(file&&file.expires_at>vaultNow())card.append(button('Vezi copia',()=>previewFile(file)));else if(item.file_id)card.append(el('p','Copia nu este disponibilă aici. Progresul originalului este păstrat.','small muted'));
   if(['analyzed','uploaded','ready','needs_review'].includes(item.state)){const chosen=cleanupCheck('Selectează '+item.name,false,job.mode!=='local'&&(!file||file.expires_at<=vaultNow())),destination=el('input');destination.value=item.destination||job.destination;destination.maxLength=200;destination.setAttribute('aria-label','Dosar pentru '+item.name);const key=jobId+':'+item.id,previous=organizerV4.edits.get(key);if(previous&&previous.item.version===item.version&&previous.item.sha256===item.sha256){chosen.input.checked=previous.chosen&&!chosen.input.disabled;destination.value=previous.destination;}const edit={item,chosen:chosen.input.checked,destination:destination.value,visible:true};organizerV4.edits.set(key,edit);chosen.input.onchange=()=>edit.chosen=chosen.input.checked;destination.oninput=()=>edit.destination=destination.value;card.append(cleanupDetails('Destinație și verificare',chosen.label,destination));}
   body.append(card);
  }
  if(data.next_cursor)body.append(button('Mai multe rezultate',()=>openOrganizerJobItems(jobId,body,data.next_cursor),'organizer-more'));
  if(!after){const controls=cleanupDetails('Mutările selectate'),blocked=['paused','cancelled','access_needed'].includes(job.state),confirmMove=cleanupCheck('Aprob mutarea originalelor selectate pe telefon.',false,blocked),approve=button('Aprobă mutările',()=>approveOrganizerItems(jobId,body,confirmMove.input.checked));controls.classList.add('organizer-review-actions');approve.disabled=blocked;controls.append(confirmMove.label,approve);if(blocked)controls.append(el('p','Continuă lucrarea pentru a aproba mutările.','small muted'));const ai=cleanupCheck('Permit analiza online a conținutului selectat cu Cloudflare AI.',false);if(job.mode==='online'&&job.ai_consent)controls.append(ai.label,button('Propune dosare cu AI',()=>analyzeOrganizerItems(jobId,body,ai.input.checked)));body.append(controls);const manage=cleanupDetails('Opțiuni lucrare',button('Șterge progresul lucrării',()=>deleteOrganizerJob(jobId)));manage.classList.add('organizer-manage');body.append(manage);}else for(const controls of body.querySelectorAll(':scope>.organizer-review-actions,:scope>.organizer-manage'))body.append(controls);
 }catch(e){if(auth===current&&device===organizerV4.device&&serial===organizerV4.itemsSerial.get(jobId)&&body.isConnected&&e.name!=='AbortError')body.append(el('p',e.message,'notice'));}
}
function selectedOrganizerItems(jobId){return [...organizerV4.edits].filter(([key,value])=>key.startsWith(jobId+':')&&value.chosen&&value.visible).map(([,value])=>value);}
async function approveOrganizerItems(jobId,body,confirmed){
 const current=auth,device=organizerV4.device,job=organizerV4.jobs.get(jobId),items=selectedOrganizerItems(jobId);if(['paused','cancelled','access_needed'].includes(job?.state))throw Error('Continuă lucrarea pentru a aproba mutările.');if(!confirmed||!items.length||items.length>100)throw Error('Selectează 1–100 de fișiere și confirmă mutarea.');if(!organizerDevice()?.grant.organize)throw Error('Permite mutările din aplicația telefonului.');
 const approved=items.map(e=>({id:e.item.id,destination:e.destination.trim()}));if(approved.some(e=>!e.destination))throw Error('Alege destinația fiecărui fișier.');
 const fingerprint=JSON.stringify([device,jobId,approved]),key='approve:'+jobId;let request=organizerV4.requests.get(key);if(request?.fingerprint!==fingerprint){request={fingerprint,request_id:crypto.randomUUID(),revision:job.revision};organizerV4.requests.set(key,request);}
 if(organizerV4.busy.has(jobId))return;organizerV4.busy.add(jobId);organizerV4.serial++;
 try{const next=await api(organizerApi(device,jobId)+'/approve',{method:'POST',body:JSON.stringify({request_id:request.request_id,revision:request.revision,items:approved,confirm:true})});if(auth!==current||device!==organizerV4.device)return;organizerV4.requests.delete(key);organizerV4.serial++;organizerV4.jobs.set(next.id,next);renderOrganizerJobs();if(body.isConnected)await openOrganizerJobItems(jobId,body);}
 catch(error){if(auth===current&&device===organizerV4.device&&error.status>=400&&error.status<500){organizerV4.requests.delete(key);await refreshOrganizerJobs();}throw error;}
 finally{if(auth===current&&device===organizerV4.device)organizerV4.busy.delete(jobId);}
}

async function analyzeOrganizerItems(jobId,body,consent){
 const current=auth,device=organizerV4.device,items=selectedOrganizerItems(jobId);if(!consent||!items.length||items.length>5)throw Error('Selectează 1–5 fișiere și permite analiza online.');
 if(organizerV4.busy.has(jobId))return;organizerV4.busy.add(jobId);try{await api('/insights/api/organizer-analysis',{method:'POST',body:JSON.stringify({device,job:jobId,ids:items.map(e=>e.item.id),consent:true})});if(auth!==current||device!==organizerV4.device||!body.isConnected)return;await openOrganizerJobItems(jobId,body);await refreshOrganizerJobs();}finally{if(auth===current&&device===organizerV4.device)organizerV4.busy.delete(jobId);}
}

async function deleteOrganizerJob(jobId){
 if(!confirm('Ștergi progresul acestei lucrări? Originalele rămân pe telefon, iar copiile online expiră separat.'))return;
 const current=auth,device=organizerV4.device;organizerV4.serial++;
 await api(organizerApi(device,jobId),{method:'DELETE'});if(auth!==current||device!==organizerV4.device)return;
 organizerV4.serial++;organizerV4.jobs.delete(jobId);renderOrganizerJobs();await refreshOrganizerJobs();
}

function organizerEvidence(analysis){
 const detail=cleanupDetails('Detaliile analizei'),coverage=analysis.coverage||{};
 if(Number.isSafeInteger(coverage.pages_processed)&&Number.isSafeInteger(coverage.pages_total))detail.append(el('p',coverage.pages_processed+' din '+coverage.pages_total+' pagini analizate','small muted'));
 if(coverage.status==='partial')detail.append(el('p','Conținut citit parțial.','small muted'));if(coverage.status==='unavailable')detail.append(el('p','Conținutul nu a putut fi citit.','small muted'));
 const evidence=Array.isArray(analysis.evidence)?analysis.evidence:[];
 for(const proof of evidence.slice(0,6)){const text=proof.kind==='text'?proof.quote:proof.observation;if(!text)continue;const row=el('div',undefined,'organizer-evidence');row.append(el('p',proof.kind==='visual'?'Observație AI · de verificat':Number.isSafeInteger(proof.page)?'Pagina '+proof.page:'Fragment citit','small muted'),el('p',text,'small'));detail.append(row);}
 const deletion=analysis.deletion;
 if(deletion?.suggested===true&&deletion.review_only===true&&deletion.requires_confirmation===true){const refs=new Set(deletion.evidence_ids||[]),linked=evidence.filter(e=>refs.has(e.id));if(linked.length){detail.append(el('h3','De verificat pentru ștergere'),el('p',deletion.reason||'Propunere AI care necesită verificare.','small'),el('p','Propunere AI · nimic nu a fost șters.','small muted'));}}
 return detail;
}
