const organizerUI={owner:null,devices:[],dirty:false,run:null,items:[],cursor:null,serial:0,edits:new Map()};
function clearOrganizer(){organizerUI.serial++;Object.assign(organizerUI,{owner:null,devices:[],dirty:false,run:null,items:[],cursor:null,edits:new Map()});for(const id of ['organizer-requests','organizer-review','organizer-plans'])$(id).replaceChildren();}
function refreshOrganizer(data){
 if(organizerUI.owner!==auth){clearOrganizer();organizerUI.owner=auth;}
 organizerUI.devices=data.devices;
 if(!organizerUI.dirty){$('organizer-requests').replaceChildren();for(const d of data.devices)$('organizer-requests').append(organizerForm(d));if(!data.devices.length)empty($('organizer-requests'),'Conectează telefonul','În aplicație: Curățenie → Configurare → Organizare din laptop.');}
 if(organizerUI.run)refreshOrganizerPlans(organizerUI.run.device);
}
function organizerForm(d){
 const form=el('form',undefined,'cleanup-device');form.append(el('h3',d.label||'Telefon'));
 if(![2,3].includes(d.protocol)||!d.grant.enabled){form.append(el('p','Actualizează aplicația și activează „Organizare din laptop” pe telefon.'));return form;}
 form.addEventListener('input',()=>organizerUI.dirty=true);
 const phrase=el('input');phrase.placeholder='Ex.: ultimele 50 de poze';phrase.maxLength=100;
 const photos=cleanupCheck('Galerie',d.grant.photos,!d.grant.photos),files=cleanupCheck('Dosarul ales',false,!d.grant.files);
 function counter(label){const wrap=el('div'),select=el('select'),custom=el('input');select.setAttribute('aria-label',label);custom.setAttribute('aria-label',label+' · număr ales');for(const[value,text]of [['50','Ultimele 50'],['100','Ultimele 100'],['custom','Număr ales'],['0','Toate cele accesibile']]){const o=el('option',text);o.value=value;select.append(o);}custom.type='number';custom.min=1;custom.max=15000;custom.value=50;custom.hidden=true;select.onchange=()=>custom.hidden=select.value!=='custom';wrap.append(select,custom);return {wrap,value:()=>select.value==='custom'?Number(custom.value):Number(select.value),set(n){select.value=['50','100','0'].includes(String(n))?String(n):'custom';custom.value=n;custom.hidden=select.value!=='custom';}};}
 const pc=counter('Număr de fotografii'),fc=counter('Număr de fișiere'),pf=el('input'),ff=el('input'),from=el('input'),to=el('input');pf.placeholder='Toată galeria sau DCIM/Camera';ff.placeholder='Tot dosarul sau Educație/Fișe';pf.maxLength=ff.maxLength=200;from.type=to.type='date';
 const recursive=cleanupCheck('Include subdosarele',true),wifi=cleanupCheck('Transfer doar prin Wi-Fi',true);
 const use=button('Folosește selecția',()=>{const q=phrase.value.toLocaleLowerCase('ro-RO');const n=/\btoate\b/.test(q)?0:Number(q.match(/\b(\d{1,5})\b/)?.[1]);if(!Number.isInteger(n)||n<0||n>15000){notice('Scrie, de exemplu, „ultimele 100 de poze”.');return;}if(/poz|foto/.test(q)&&d.grant.photos){photos.input.checked=true;files.input.checked=false;pc.set(n);}else if(/document|fi[sș]ier/.test(q)&&d.grant.files){photos.input.checked=false;files.input.checked=true;fc.set(n);}else{notice('Alege o sursă activată: poze sau fișiere.');return;}organizerUI.dirty=true;});use.type='button';
 const grid=el('div',undefined,'cleanup-grid');grid.append(photos.label,cleanupField('Fotografii',pc.wrap),files.label,cleanupField('Fișiere',fc.wrap));form.append(grid);
 const mode=el('select');mode.setAttribute('aria-label','Mod de organizare');for(const [v,t]of [['manual','Manual · aleg destinațiile'],['local','Sugestii locale · pe telefon']]){const o=el('option',t);o.value=v;mode.append(o);}mode.value='local';form.append(cleanupField('Organizare',mode));
 const filters=el('div',undefined,'cleanup-grid');filters.append(cleanupField('Album sau cale în Galerie',pf),cleanupField('Subdosar',ff),cleanupField('De la',from),cleanupField('Până la',to),recursive.label,wifi.label);
 form.append(cleanupDetails('Personalizează selecția',cleanupField('Selecție rapidă',phrase),use,filters,el('p','Perioada și dosarul se aplică înainte de alegerea celor mai recente. Fotografiile folosesc data fotografierii, apoi data modificării; fișierele folosesc data modificării. Datele includ zilele alese, în fusul orar al laptopului.','muted small'),el('p','„Toate”: până la 15.000 de elemente pe sursă. Transfer: maximum 500 de copii / 512 MB, până la 25 MB per fișier.','muted small')));
 const submit=el('button','Pregătește selecția','primary');submit.type='submit';form.append(el('p','Primești copii. Originalele rămân pe telefon.','muted small'),submit);let id=null;
 form.onsubmit=async e=>{e.preventDefault();const current=auth;submit.disabled=true;
  try{const end=to.value?new Date(to.value+'T00:00:00'):null;if(end)end.setDate(end.getDate()+1);const selection={mode:mode.value,photos:photos.input.checked,files:files.input.checked,photo_count:pc.value(),file_count:fc.value(),photo_folder:pf.value.trim(),file_folder:ff.value.trim(),from:from.value?new Date(from.value+'T00:00:00').getTime():null,to:end?.getTime()??null,recursive:recursive.input.checked,wifi_only:wifi.input.checked};
   const payload=JSON.stringify(selection);if(!id||id.payload!==payload)id={value:crypto.randomUUID(),payload};await api(`/v2/organizer/devices/${d.id}/request`,{method:'POST',body:JSON.stringify({id:id.value,revision:d.revision,selection})});if(auth!==current)return;organizerUI.dirty=false;notice('Cerere trimisă. Așteaptă conectarea telefonului; în fundal poate dura mai mult.',true);await refreshCleanup(true);
  }catch(error){if(auth===current&&error.name!=='AbortError')notice(error.message);}finally{if(auth===current)submit.disabled=false;}};
 return form;
}
async function openOrganizer(run,after=null){
 const current=auth,serial=++organizerUI.serial;organizerUI.run=run;organizerUI.edits.clear();$('organizer-review').replaceChildren(el('p','Încarc rezultatele…'));
 try{const data=await api(`/v2/organizer/devices/${run.device}/runs/${run.id}/items`+(after?'?after='+encodeURIComponent(after):''));if(auth!==current||serial!==organizerUI.serial)return;organizerUI.items=data.items;organizerUI.cursor=data.next_cursor;renderOrganizer(data.total);refreshOrganizerPlans(run.device);}catch(e){if(auth===current&&serial===organizerUI.serial){$('organizer-review').replaceChildren(el('p',e.message));}}
}
function renderOrganizer(total){
 const box=$('organizer-review'),r=organizerUI.run;box.replaceChildren();box.append(el('h3','Alege destinațiile'),el('p',total+' rezultate · '+dt(r.at),'muted small'));
 if(!total){box.append(el('p','Așteptăm rezultatele de pe telefon.'));box.append(button('Actualizează',()=>openOrganizer(r)));return;}
 const table=el('table'),head=el('tr');for(const s of ['Alege','Fișier / conținut','Dosar destinație'])head.append(el('th',s));table.append(head);
 for(const item of organizerUI.items){const row=el('tr'),check=el('input');check.type='checkbox';check.disabled=!item.received;check.setAttribute('aria-label','Selectează '+item.name);const cell=el('td');cell.append(check);row.append(cell);const details=el('td');details.append(el('strong',item.name),el('p',(item.source==='photos'?'Galerie':'Dosar')+' · '+item.folder,'muted small'));if(item.partial)details.append(el('p','Analiză parțială · de verificat','muted small'));if(item.reason||item.excerpt){const d=cleanupDetails('Detalii');if(item.reason)d.append(el('p',item.reason,'muted small'));if(item.excerpt)d.append(el('pre',item.excerpt));details.append(d);}if(item.received)details.append(button('Vezi copia',()=>previewFile(item)));else details.append(el('p','Copia este în așteptare.','muted small'));row.append(details);
  const destination=el('input');destination.value=item.destination;destination.maxLength=200;destination.setAttribute('aria-label','Dosar pentru '+item.name);const reason=el('p',r.selection?.mode==='manual'?'Alegi tu':'Sugestie locală','muted small'),target=el('td');target.append(destination,reason);row.append(target);table.append(row);organizerUI.edits.set(item.id,{item,check,destination,reason});
 }
 const wrap=el('div',undefined,'organizer-table');wrap.append(table);box.append(wrap);
 const selectionRow=el('div',undefined,'row');for(const[source,label]of [['photos','Selectează Galeria'],['files','Selectează dosarul']])selectionRow.append(button(label,()=>{for(const e of organizerUI.edits.values())e.check.checked=e.item.source===source&&!e.check.disabled;}));box.append(selectionRow);
 const online=cleanupCheck('Permit analiza online a selecției',false);
 const ai=button('Analizează cu AI',async()=>{
  const selected=[...organizerUI.edits].filter(([,e])=>e.check.checked),current=auth,serial=organizerUI.serial;if(!online.input.checked||!selected.length||selected.length>5){notice('Selectează 1–5 fișiere și permite analiza online.');return;}ai.disabled=true;
  try{const result=await api('/insights/api/organize',{method:'POST',body:JSON.stringify({device:r.device,run:r.id,ids:selected.map(([id])=>id),consent:true})});if(auth!==current||serial!==organizerUI.serial)return;for(const v of result.items){const e=organizerUI.edits.get(v.id);e.destination.value=v.destination;e.reason.textContent='AI online · '+({low:'de verificat',medium:'încredere medie',high:'încredere ridicată'}[v.confidence])+': '+v.reason;}notice('Propuneri actualizate. Verifică-le înainte de a aproba mutarea.',true);}catch(e){if(auth===current)notice(e.message);}finally{if(auth===current)ai.disabled=false;}
 });box.append(cleanupDetails('Sugestii AI online · opțional',el('p','Cloudflare procesează textul extras și miniaturile celor 1–5 fișiere selectate.','muted small'),online.label,ai,el('p','Llama 70B și analiză vizuală · maximum 30 de inferențe pe zi. Fiecare miniatură folosește o inferență vizuală, plus una pentru grupare.','muted small')));
 const confirm=cleanupCheck('Am verificat destinațiile și aprob mutarea fișierelor originale pe telefon.',false);box.append(confirm.label);let planRequest=null;
 const apply=button('Mută originalele pe telefon',async()=>{
  const entries=[...organizerUI.edits.values()].filter(e=>e.check.checked),current=auth;if(!confirm.input.checked||!entries.length){notice('Selectează fișiere și confirmă mutarea originalelor.');return;}if(new Set(entries.map(e=>e.item.source)).size!==1){notice('Aplică separat fotografiile din Galerie și fișierele din dosar.');return;}
  const d=organizerUI.devices.find(d=>d.id===r.device);if(!d?.grant.organize){notice('Activează „Aplică mutările aprobate din laptop” în configurarea telefonului. Apoi cere o analiză nouă.');return;}apply.disabled=true;
  try{const items=entries.map(e=>({id:e.item.id,destination:e.destination.value.trim()})),fingerprint=JSON.stringify(items);if(!planRequest||planRequest.fingerprint!==fingerprint)planRequest={id:crypto.randomUUID(),fingerprint};const p=await api(`/v2/organizer/devices/${r.device}/runs/${r.id}/plan`,{method:'POST',body:JSON.stringify({id:planRequest.id,grant_id:d.grant.id,items,confirm:true})});if(auth!==current)return;notice(p.message,true);await refreshOrganizerPlans(r.device);}catch(e){if(auth===current)notice(e.message);}finally{if(auth===current)apply.disabled=false;}
 });apply.classList.add('primary');box.append(apply,el('p','Galeria și dosarul se mută separat. Fotografiile cer și confirmarea Android pe telefon.','muted small'),cleanupDetails('Despre mutare',el('p','Documentele rămân în dosarul autorizat. Pentru Galerie, deschide Curățenie pe telefon și confirmă planul. Rezultatul apare după răspunsul telefonului. Nu se șterg automat fișiere; jurnalul de anulare păstrează ultima operație.','muted small')));
 if(organizerUI.cursor)box.append(button('Pagina următoare (lasă editările nesalvate)',()=>openOrganizer(r,organizerUI.cursor)));box.append(button('Reîncarcă prima pagină',()=>openOrganizer(r)));
 box.scrollIntoView({behavior:'smooth',block:'start'});
}
async function refreshOrganizerPlans(device){
 const current=auth;try{const data=await api(`/v2/organizer/devices/${device}/plans`);if(auth!==current||organizerUI.run?.device!==device)return;const box=$('organizer-plans');box.replaceChildren();for(const p of data.plans){const row=el('article',undefined,'cleanup-result');row.append(el('h3',({approved:'În așteptarea telefonului',awaiting_phone:'Confirmă pe telefon',applying:'Se mută pe telefon',complete:'Rezultat primit',error:'Necesită atenție'}[p.state])||p.state),el('p',p.message),el('p',`${p.moved} mutate · ${p.copied} copiate · ${p.skipped} nemodificate`,'muted small'));if(['approved','awaiting_phone'].includes(p.state))row.append(button('Anulează planul',async()=>{try{await api(`/v2/organizer/devices/${device}/plans/${p.id}`,{method:'DELETE'});refreshOrganizerPlans(device);}catch(e){notice(e.message);}}));box.append(row);}}catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}
}
