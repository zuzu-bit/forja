const phoneDurations=new Map(),phoneWindows=new Map();
function localTimeInput(at){const d=new Date(at);return new Date(at-d.getTimezoneOffset()*60000).toISOString().slice(0,16);}
async function refreshPhones(){
  try{
    const data=await api('/insights/api/phones');
    if(document.activeElement?.matches('#phone-list input,#phone-list select'))return;
    $('phone-list').replaceChildren();
    if(!data.phones.length)empty($('phone-list'),'Asociază telefonul o singură dată.','În FORJA: Profil → Permisiuni și sincronizare → Control din contul web.');
    for(const p of data.phones){
      const card=el('article',undefined,'panel');card.append(el('h2',p.name));
      const confirmed=p.command&&p.handled_command===p.command.id;
      const future=p.command?.action==='start'&&!confirmed&&p.command.start_at>Date.now();
      const waiting=p.command&&!confirmed&&(p.command.action==='stop'||p.command.start_before>Date.now());
      const missed=p.command?.action==='start'&&!confirmed&&p.command.start_before<=Date.now();
      const label=future?'Interval programat · așteaptă ora de început':!p.online?'Telefon indisponibil · ultima conectare '+dt(p.seen_at):waiting?'Comandă trimisă · aștept confirmarea telefonului':missed?'Interval ratat · telefonul nu a confirmat pornirea':p.state==='recording'?'Înregistrare confirmată de telefon':p.state==='start_failed'?'Pornirea nu a reușit pe telefon':'Telefon conectat';
      card.append(el('p',label,'campaign-status'));
      if(p.audio_status)card.append(el('p',p.audio_status,'small muted'));
      const transfer=p.recording_transfer;
      if(transfer){
        const labels={recording:'Se înregistrează pe telefon',pending:'Audio salvat · așteaptă trimiterea',uploading:'Se încarcă audio pe server',uploaded:'Telefonul a confirmat trimiterea audio',failed:'Audio păstrat pe telefon · trimitere nereușită',cancelled:'Trimitere anulată',interrupted:'Înregistrare întreruptă'};
        card.append(el('p',labels[transfer.state]||transfer.state,'campaign-status'));
        if(transfer.message)card.append(el('p',transfer.message,'small muted'));
        if(transfer.pending_count)card.append(el('p',transfer.pending_count+' înregistrări așteaptă trimiterea.','small muted'));
      }
      card.append(button('Vezi și ascultă înregistrările',async()=>{page('data');document.querySelector('[data-tab="audio"]').click();await refresh();}));
      if(p.command?.action==='start')card.append(el('p',dt(p.command.start_at)+' → '+dt(p.command.stop_at),'small muted'));
      const send=async value=>{await api('/insights/api/phones/'+p.id+'/command',{method:'POST',body:JSON.stringify({id:crypto.randomUUID(),revision:p.revision,...value})});await refreshPhones();};
      const minutes=el('select');minutes.setAttribute('aria-label','Durata înregistrării');
      for(const m of [5,10,15,30,60]){const o=el('option',m+' minute');o.value=m;minutes.append(o);}
      minutes.value=phoneDurations.get(p.id)||'10';minutes.addEventListener('change',()=>phoneDurations.set(p.id,minutes.value));
      const row=el('div',undefined,'row');
      row.append(minutes,button('Pornește acum',()=>send({action:'start',minutes:Number(minutes.value)}),'primary'),
        button('Oprește înregistrarea',()=>send({action:'stop',minutes:1})),
        button(p.collection_enabled?'Oprește colectarea autorizată':'Pornește colectarea autorizată',async()=>{
          await api('/insights/api/phones/'+p.id+'/collection',{method:'POST',body:JSON.stringify({enabled:!p.collection_enabled,revision:p.revision})});await refreshPhones();
        }));
      card.append(row);
      const schedule=el('details');schedule.style.marginTop='20px';schedule.append(el('summary','Alege intervalul de la ora X la ora Y'));
      const saved=phoneWindows.get(p.id)||{start:localTimeInput(Math.ceil(Date.now()/60000)*60000+60000),end:localTimeInput(Math.ceil(Date.now()/60000)*60000+31*60000)};
      schedule.open=!!saved.open;schedule.addEventListener('toggle',()=>{saved.open=schedule.open;phoneWindows.set(p.id,saved);});
      const fields=el('div',undefined,'row'),start=el('input'),end=el('input');
      for(const [input,key,title] of [[start,'start','De la'],[end,'end','Până la']]){
        input.type='datetime-local';input.value=saved[key];input.id='phone-'+p.id+'-'+key;
        const field=el('div',undefined,'field'),label=el('label',title);label.htmlFor=input.id;
        input.addEventListener('input',()=>{saved[key]=input.value;phoneWindows.set(p.id,saved);});field.append(label,input);fields.append(field);
      }
      fields.append(button('Programează intervalul',async()=>{
        const start_at=new Date(start.value).getTime(),stop_at=new Date(end.value).getTime();
        if(!Number.isFinite(start_at)||!Number.isFinite(stop_at)||start_at<Date.now()-30000||start_at>Date.now()+86400000||stop_at-start_at<60000||stop_at-start_at>3600000)throw Error('Alege în următoarele 24 de ore un interval de 1–60 de minute.');
        await send({action:'start',minutes:Math.ceil((stop_at-start_at)/60000),start_at,stop_at});
      }));
      schedule.append(fields,el('p','Orele sunt în fusul orar al browserului. FORJA trebuie să fie deschisă pe telefon la începutul intervalului; o pornire ratată nu este recuperată mai târziu.','small muted'));
      card.append(schedule,el('p','După pornire, înregistrarea poate continua cu ecranul blocat. Android afișează indicatorul și notificarea cu Oprește. La final, un singur fișier se trimite automat în contul tău.','small muted'));
      $('phone-list').append(card);
    }
  }catch(e){if(e.name!=='AbortError')notice(e.message);}
}
setInterval(()=>{if(auth&&!document.hidden&&!$('page-control').hidden)refreshPhones();},10000);
