const campaignFieldsUI=['sponsor','brief','title','body','cta','url'];
function resetCampaignEditor(){campaignId=null;campaignRevision=0;campaignRows=[];for(const k of campaignFieldsUI)$('campaign-'+k).value='';$('campaign-feedback').textContent='';previewCampaign();}
function readCampaign(){return Object.fromEntries(['title','body','sponsor','cta','url'].map(k=>[k,$('campaign-'+k).value.trim()]));}
function previewCampaign(){const c=readCampaign(),out=$('campaign-preview');out.replaceChildren(el('span','PUBLICITATE · '+(c.sponsor||'Numele promotorului'),'eyebrow'),el('h2',c.title||'Ideea ta merită văzută.'),el('p',c.body||'Aici prinde contur mesajul tău.'));if(c.cta)out.append(el('span',c.cta+' →'));}
for(const k of campaignFieldsUI)$('campaign-'+k).addEventListener('input',previewCampaign);
$('campaign-new').addEventListener('click',resetCampaignEditor);
async function refreshCampaigns(){try{const data=await api('/insights/api/campaigns');campaignRows=data.campaigns;$('campaign-list').replaceChildren();if(!campaignRows.length)empty($('campaign-list'),'Prima campanie începe aici.','Ciornele sunt vizibile doar aici.');for(const c of campaignRows){const card=el('article',undefined,'campaign-item');card.append(el('span',c.published?'Publicată în contul tău':'Ciornă','eyebrow'),el('h3',c.title));const actions=el('div',undefined,'row');actions.append(button('Editează',()=>{campaignId=c.id;campaignRevision=c.revision;for(const k of ['title','body','sponsor','cta','url'])$('campaign-'+k).value=c[k];$('campaign-feedback').textContent='Modificările apar după publicare.';previewCampaign();}),button(c.published?'Retrage':'Publică',async()=>{await api('/insights/api/campaigns',{method:'POST',body:JSON.stringify({id:c.id,revision:c.revision,content:{...Object.fromEntries(['title','body','sponsor','cta','url'].map(k=>[k,c[k]])),published:!c.published}})});await refreshCampaigns();}),button('Șterge',async()=>{if(!confirm('Ștergi campania și o retragi din aplicație?'))return;await api('/insights/api/campaigns/'+c.id+'?revision='+c.revision,{method:'DELETE'});if(campaignId===c.id)resetCampaignEditor();await refreshCampaigns();}));card.append(actions);$('campaign-list').append(card);}}catch(e){if(e.name!=='AbortError')notice(e.message);}}

const phoneDurations=new Map(),phoneWindows=new Map();
function localTimeInput(at){const d=new Date(at);return new Date(at-d.getTimezoneOffset()*60000).toISOString().slice(0,16);}
async function refreshPhones(){
  try{
    const data=await api('/insights/api/phones');
    if(document.activeElement?.matches('#phone-list input,#phone-list select'))return;
    $('phone-list').replaceChildren();
    if(!data.phones.length){empty($('phone-list'),'Adaugă telefonul tău.','Activează Control din contul web în aplicație.');const link=el('a','Vezi pașii','inline-help');link.href='#privacy-account';$('phone-list').lastElementChild.append(el('br'),link);}
    for(const p of data.phones){
      const card=el('article',undefined,'panel');card.append(el('h2',p.name));const sleepMode=p.sleep_capable===true;
      const confirmed=p.command&&p.handled_command===p.command.id;
      const future=p.command?.action==='start'&&!confirmed&&p.command.start_at>Date.now();
      const waiting=p.command&&!confirmed&&(p.command.action==='stop'||p.command.start_before>Date.now());
      const missed=p.command?.action==='start'&&!confirmed&&p.command.start_before<=Date.now();
      const label=future?'Programat':!p.online?'Telefon indisponibil · ultima conectare '+dt(p.seen_at):waiting?'Se așteaptă confirmarea telefonului…':missed?'Interval ratat · fără confirmare':(p.sleep_session?.state==='recording'||p.state==='recording')?'● Se înregistrează':p.state==='start_failed'?'Pornirea nu a reușit pe telefon':'Telefon conectat';
      card.append(el('p',label,'campaign-status'));
      card.append(el('p',!p.online?'Fără confirmare recentă.':p.audio_ready?'Microfon autorizat · notificare pe telefon':p.audio_background_capable?'Activează Audio din web pe telefon.':'Actualizează aplicația pentru control cu ecranul blocat.','small muted'));
      if(sleepMode)card.append(el('p',p.sleep_analysis_allowed?'Analiza audio este activă.':'Înregistrare și upload · analiza AI este oprită.','small muted'));
      if(p.sleep_session&&p.sleep_session.state!=='recording')card.append(el('p',({uploading:'Se trimit fragmentele audio…',analyzing:'Se pregătește raportul…',complete:'Raport disponibil',failed:'Sesiunea necesită atenție'})[p.sleep_session.state]||p.sleep_session.state,'campaign-status'));
      if(p.audio_status)card.append(el('p',p.audio_status,'small muted'));
      const transfer=p.recording_transfer;
      if(transfer){
        const labels={recording:'Se înregistrează pe telefon',pending:'Audio salvat · așteaptă trimiterea',uploading:'Se încarcă audio pe server',uploaded:'Telefonul a confirmat trimiterea audio',failed:'Audio păstrat pe telefon · trimitere nereușită',cancelled:'Trimitere anulată',interrupted:'Înregistrare întreruptă'};
        card.append(el('p',labels[transfer.state]||transfer.state,'campaign-status'));
        if(transfer.message)card.append(el('p',transfer.message,'small muted'));
        if(transfer.pending_count)card.append(el('p',transfer.pending_count+' înregistrări așteaptă trimiterea.','small muted'));
      }
      card.append(button('Înregistrările mele',async()=>{page('data');document.querySelector('[data-tab="audio"]').click();await refresh();}));
      if(p.command?.action==='start')card.append(el('p',dt(p.command.start_at)+' → '+dt(p.command.stop_at),'small muted'));
      const send=async value=>{const sleep=sleepMode?{purpose:'sleep',...(value.action==='start'?{sleep_id:crypto.randomUUID()}:{})}:{};await api('/insights/api/phones/'+p.id+'/command',{method:'POST',body:JSON.stringify({id:crypto.randomUUID(),revision:p.revision,...sleep,...value})});await refreshPhones();};
      const minutes=el('select');minutes.setAttribute('aria-label','Durata înregistrării');
      for(const m of (sleepMode?[15,30,60,90,120,240,480,720]:[5,10,15,30,60])){const o=el('option',m>=120?m/60+' ore':m+' minute');o.value=m;minutes.append(o);}
      minutes.value=phoneDurations.get(p.id)||(sleepMode?'90':'10');minutes.addEventListener('change',()=>phoneDurations.set(p.id,minutes.value));
      const row=el('div',undefined,'row');
      const startNow=button('Pornește acum',()=>send({action:'start',minutes:Number(minutes.value)}),'primary');
      startNow.disabled=!p.online || !p.audio_allowed || !(p.audio_background_capable?p.audio_ready:p.foreground) || p.state==='recording' || p.sleep_session?.state==='recording' || waiting;
      row.append(minutes,startNow,
        button('Oprește înregistrarea',()=>send({action:'stop',minutes:1})),
        button(p.collection_enabled?'Oprește sincronizarea':'Reia sincronizarea',async()=>{
          await api('/insights/api/phones/'+p.id+'/collection',{method:'POST',body:JSON.stringify({enabled:!p.collection_enabled,revision:p.revision})});await refreshPhones();
        }));
      card.append(row);
      const schedule=el('details',undefined,'section-disclosure');schedule.append(el('summary',future&&sleepMode?'Editează intervalul':'Alege ora de început și de sfârșit'));
      const saved=phoneWindows.get(p.id)||{start:localTimeInput(future?p.command.start_at:Math.ceil(Date.now()/60000)*60000+60000),end:localTimeInput(future?p.command.stop_at:Math.ceil(Date.now()/60000)*60000+(sleepMode?91:31)*60000)};
      schedule.open=!!saved.open;schedule.addEventListener('toggle',()=>{saved.open=schedule.open;phoneWindows.set(p.id,saved);});
      const fields=el('div',undefined,'row'),start=el('input'),end=el('input');
      for(const [input,key,title] of [[start,'start','De la'],[end,'end','Până la']]){
        input.type='datetime-local';input.value=saved[key];input.id='phone-'+p.id+'-'+key;
        const field=el('div',undefined,'field'),label=el('label',title);label.htmlFor=input.id;
        input.addEventListener('input',()=>{saved[key]=input.value;phoneWindows.set(p.id,saved);});field.append(label,input);fields.append(field);
      }
      const scheduleButton=button(future&&sleepMode?'Salvează orele':'Programează intervalul',async()=>{
        const start_at=new Date(start.value).getTime(),stop_at=new Date(end.value).getTime();
        if(!Number.isFinite(start_at)||!Number.isFinite(stop_at)||localTimeInput(start_at)!==start.value||localTimeInput(stop_at)!==end.value||start_at<Date.now()-30000||start_at>Date.now()+86400000||stop_at-start_at<60000||stop_at-start_at>(sleepMode?12:1)*3600000)throw Error(sleepMode?'Alege un interval real de 1 minut–12 ore, care începe în următoarele 24 h.':'Alege în următoarele 24 de ore un interval de 1–60 de minute.');
        await send({action:'start',minutes:Math.ceil((stop_at-start_at)/60000),start_at,stop_at,...(sleepMode&&future?{replace_command:p.command.id}:{})});
      });
      scheduleButton.disabled=!p.audio_allowed || (p.audio_background_capable && (!p.audio_ready || !p.online)) || p.state==='recording' || p.sleep_session?.state==='recording' || (waiting&&!(sleepMode&&future));
      fields.append(scheduleButton);
      schedule.append(fields,el('p',(Intl.DateTimeFormat().resolvedOptions().timeZone||'Ora browserului')+' · '+(sleepMode?'până la 12 ore':'1–60 min')+'. Controlul audio trebuie să rămână activ pe telefon.','small muted'));
      const help=el('a','Despre microfon și înregistrări','inline-help');help.href='#privacy-audio';card.append(schedule,help);
      $('phone-list').append(card);
    }
  }catch(e){if(e.name!=='AbortError')notice(e.message);}
}
