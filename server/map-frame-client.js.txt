/* The 2D map is usable independently of WebGL and the vector provider. */
(function(){
 'use strict';
 const status=document.getElementById('map-status'),toggle=document.getElementById('map-mode'),surface=document.getElementById('vector');
 const send=(type,value={})=>{window.ForjaMapHost?.event(JSON.stringify({type,...value}));};
 const message=text=>{status.textContent=text;status.hidden=!text;};
 const report=(stage,error)=>{const detail=String(error?.message||error||'Unknown map error');status.dataset.error=stage+': '+detail;console.error('[FORJA map] '+stage,detail);};
 let base,tiles,overlay,vector=null,vectorReady=false,mode3d=false,loading=false,ready=false,disposed=false,data={},generation=0;
 function modeState(){toggle.textContent=loading?'…':mode3d?'2D':'3D';toggle.disabled=loading;toggle.setAttribute('aria-pressed',String(mode3d));send('mode',{mode:mode3d?'3d':'2d',loading});}
 function fallback(error){if(vectorReady)try{const p=vector.getCenter();base.setView([p.lat,p.lon],p.zoom||base.getZoom(),{animate:false});}catch{}generation++;loading=false;mode3d=false;vectorReady=false;surface.hidden=true;surface.style.opacity='0';surface.style.pointerEvents='none';try{vector?.destroy();}catch{}vector=null;surface.replaceChildren();if(error){report('3D',error);message('3D nu este disponibil acum. Harta 2D rămâne activă.');}modeState();send('error',{recoverable:true,message:status.textContent});}
 function point(feature,latlng){const kind=feature.properties?.kind||'place',badge=document.createElement('span');badge.className='journey-pin';badge.style.setProperty('--pin-color',kind==='visit'?'#F3B952':kind==='person'?'#c9e89c':kind==='device'?'#cbd2d8':'#f6f6ed');badge.textContent=kind==='visit'?'★':kind==='person'?(feature.properties?.label||'●'):'●';return L.marker(latlng,{title:feature.properties?.name||'',icon:L.divIcon({html:badge,className:'',iconSize:kind==='person'?[36,36]:[25,25]})});}
 function paint(){if(!overlay)return;overlay.clearLayers();for(const key of ['zones','routes','visits','people','places','devices']){const collection=data[key];if(!collection?.features?.length)continue;L.geoJSON(collection,{style:key==='routes'?{color:'#4F92FF',weight:4,opacity:.92}:{color:'#37875d',weight:1,fillColor:'#71C99B',fillOpacity:.26},pointToLayer:point,onEachFeature:(feature,layer)=>{if(feature.geometry?.type==='Point')layer.on('click',()=>{const p=feature.properties||{},coordinates=feature.geometry.coordinates;send('pick',{item:{kind:p.kind,id:String(p.id),name:p.name||'',lon:coordinates[0],lat:coordinates[1]}});});}}).addTo(overlay);}vector?.setData(data);}
 function set3D(enabled){
  if(disposed)return false;
  if(!enabled){generation++;loading=false;if(vectorReady){const p=vector.getCenter();base.setView([p.lat,p.lon],p.zoom||base.getZoom(),{animate:false});}try{vector?.destroy();}catch{}vector=null;vectorReady=false;surface.replaceChildren();mode3d=false;surface.hidden=true;surface.style.opacity='0';surface.style.pointerEvents='none';message('');modeState();return true;}
  if(loading)return false;
  if(vectorReady){surface.hidden=false;const p=base.getCenter();vector.resize();vector.focus({lat:p.lat,lon:p.lng},base.getZoom());vector.set3D(true);surface.style.opacity='1';surface.style.pointerEvents='auto';mode3d=true;message('');modeState();return true;}
  loading=true;modeState();message('Se încarcă 3D…');surface.hidden=false;surface.style.opacity='0';surface.style.pointerEvents='none';const request=++generation;
  try{const p=base.getCenter();vector=ForjaMapRenderer.create('vector',{center:[p.lng,p.lat],zoom:base.getZoom(),onReady:()=>{if(request!==generation||disposed)return;vectorReady=true;loading=false;vector.setData(data);vector.resize();vector.set3D(true);surface.style.opacity='1';surface.style.pointerEvents='auto';mode3d=true;message('');modeState();},onError:error=>{if(request===generation)fallback(error);},onPick:item=>send('pick',{item}),onMove:point=>{if(mode3d)send('move',{point});}});}
  catch(error){if(request===generation)fallback(error);}return true;
 }
 try{
  base=L.map('map',{zoomControl:false}).setView([45.8,24.9],6);L.control.zoom({position:'bottomright'}).addTo(base);overlay=L.layerGroup().addTo(base);
  tiles=L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png',{maxZoom:19,attribution:'&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap contributors</a>'}).addTo(base);
  tiles.on('tileload',()=>{if(!ready){ready=true;message('');send('ready',{mode:'2d'});}});
  tiles.on('tileerror',()=>{if(!ready)message('Fundalul hărții se încarcă greu. Verifică conexiunea.');});
  base.on('moveend',()=>{if(!mode3d){const p=base.getCenter();send('move',{point:{lat:p.lat,lon:p.lng,zoom:base.getZoom()}});}});
  toggle.addEventListener('click',()=>set3D(!mode3d));if(window.ForjaMapHost)toggle.hidden=true;
  window.ForjaMap={setData(value){data=value||{};paint();},set3D,focus(p,zoom=16){base.setView([p.lat,p.lon],zoom,{animate:false});vector?.focus(p,zoom);},fit(points){if(points.length)base.fitBounds(points.map(p=>[p.lat,p.lon]),{padding:[70,70],maxZoom:15});vector?.fit(points);},getCenter(){if(mode3d)return vector.getCenter();const p=base.getCenter();return {lat:p.lat,lon:p.lng,lng:p.lng,zoom:base.getZoom()};},resize(){base.invalidateSize();vector?.resize();},destroy(){disposed=true;generation++;vector?.destroy();base.remove();},get ready(){return ready;},get mode3d(){return mode3d;}};
  modeState();
 }catch(error){report('2D',error);message('Harta nu s-a încărcat. Reîncearcă deschiderea ei.');send('error',{recoverable:false,message:status.textContent});}
})();
