const cleanupUI={serial:0,dirty:false};
function clearCleanup(){cleanupUI.serial++;cleanupUI.dirty=false;$('cleanup-devices').replaceChildren();$('cleanup-runs').replaceChildren();}
function cleanupField(text,input){const label=el('label',undefined,'cleanup-field');label.append(el('span',text),input);return label;}
function cleanupCheck(text,checked,disabled=false){const input=el('input');input.type='checkbox';input.checked=checked;input.disabled=disabled;const label=cleanupField(text,input);label.classList.add('cleanup-check');return {input,label};}
function cleanupDetails(title,...content){const detail=el('details',undefined,'cleanup-details');detail.append(el('summary',title),...content);return detail;}
async function refreshCleanup(force=false){
 if(!auth)return;const current=auth,serial=++cleanupUI.serial;
 try{
  const data=await api('/v2/cleanup/devices');if(auth!==current||serial!==cleanupUI.serial)return;
  refreshOrganizer(data);
  if(force||!cleanupUI.dirty){
   cleanupUI.dirty=false;$('cleanup-devices').replaceChildren();
   if(!data.devices.length)empty($('cleanup-devices'),'Conectează telefonul','În aplicație: Curățenie → Analizează-mi automat.');
   for(const d of data.devices)$('cleanup-devices').append(cleanupDevice(d));
  }
  $('cleanup-runs').replaceChildren();
  for(const r of data.runs){
   const card=el('article',undefined,'cleanup-result');
   const phase=r.cancelled?'Oprită · program modificat':r.paused?'În pauză':({queued:'În așteptare',analyzing:'Se analizează',analyzed:'Se transferă copiile',complete:r.uploaded===r.total?'Gata':'Transfer incomplet',error:'Necesită atenție'}[r.phase]||r.phase);
   card.append(el('h3',phase),el('p',dt(r.at),'muted small'));
   const stats=el('div',undefined,'cleanup-result-stats');for(const text of [r.analyzed+' analizate',r.proposals.length+' propuneri',r.duplicates+' duplicate',r.uploaded+'/'+r.total+' copii primite'])stats.append(el('span',text,'pill'));card.append(stats);
   if(r.message)card.append(el('p',r.message,'muted small'));
   if(r.proposals.length){const detail=cleanupDetails('Propuneri · '+r.proposals.length);for(const p of r.proposals){const row=el('div',undefined,'cleanup-proposal');row.append(el('strong',p.name),el('p','→ '+p.destination),el('p',p.reason,'muted small'));detail.append(row);}card.append(detail);}
   if([2,3].includes(data.devices.find(d=>d.id===r.device)?.protocol))card.append(button('Organizează selecția',()=>openOrganizer(r)));
   $('cleanup-runs').append(card);
  }
 }catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}
}
function cleanupDevice(d){
 const form=el('form',undefined,'cleanup-device');form.addEventListener('input',()=>{cleanupUI.dirty=true;});form.addEventListener('change',()=>{cleanupUI.dirty=true;});
 form.append(el('h3',d.label||'Telefon '+d.id.slice(0,8)),el('p','Ultima conectare · '+dt(d.last_seen),'muted small'));
 if(!d.grant.enabled){form.append(el('p','Dezactivat. Pornește analiza din aplicația de pe telefon.'));return form;}
 const enabled=cleanupCheck('Analiză automată',d.schedule.enabled),photos=cleanupCheck('Fotografii',d.schedule.photos,!d.grant.photos),files=cleanupCheck('Dosarul ales',d.schedule.files,!d.grant.files),wifi=cleanupCheck('Doar conexiune fără cost de date',d.schedule.wifi_only);
 const count=el('select');count.name='count';for(const n of [25,50]){const option=el('option','Ultimele '+n+' din fiecare sursă');option.value=n;count.append(option);}count.value=d.schedule.count;
 const zone=el('input');zone.value=d.schedule.timezone;zone.maxLength=64;zone.name='timezone';
 const times=Array.from({length:4},(_,i)=>{const input=el('input');input.type='time';input.name='time'+i;input.value=d.schedule.times[i]||'';return input;});
 const grid=el('div',undefined,'cleanup-grid');grid.append(enabled.label,cleanupField('Câte analizăm?',count),photos.label,files.label,cleanupField('Ora',times[0]));
 const advanced=el('div',undefined,'cleanup-grid');advanced.append(cleanupField('Fus orar',zone),wifi.label);times.slice(1).forEach((input,i)=>advanced.append(cleanupField('Ora suplimentară '+(i+1),input)));
 const days=['Luni','Marți','Miercuri','Joi','Vineri','Sâmbătă','Duminică'].map((name,i)=>cleanupCheck(name,d.schedule.days.includes(i+1)));
 const dayRow=el('div',undefined,'cleanup-days');days.forEach(d=>dayRow.append(d.label));
 const save=el('button','Salvează programul','primary');save.type='submit';
 form.append(grid,el('p','Zile','muted small'),dayRow,cleanupDetails('Mai multe opțiuni',advanced),save);
 form.addEventListener('submit',async event=>{
  event.preventDefault();save.disabled=true;const current=auth;
  try{
   const schedule={enabled:enabled.input.checked,count:Number(count.value),photos:photos.input.checked,files:files.input.checked,times:times.map(t=>t.value).filter(Boolean),days:days.flatMap((d,i)=>d.input.checked?[i+1]:[]),timezone:zone.value.trim(),wifi_only:wifi.input.checked};
   await api('/v2/cleanup/devices/'+d.id+'/schedule',{method:'POST',body:JSON.stringify({revision:d.revision,schedule})});
   if(auth!==current)return;cleanupUI.dirty=false;notice('Program salvat. Așteaptă sincronizarea telefonului.',true);await refreshCleanup(true);
  }catch(e){if(auth===current&&e.name!=='AbortError')notice(e.message);}finally{if(auth===current)save.disabled=false;}
 });return form;
}
$('cleanup-refresh').addEventListener('click',()=>refreshCleanup(true));
setInterval(()=>{if(auth&&!document.hidden&&!$('page-files').hidden)refreshCleanup();},30000);
